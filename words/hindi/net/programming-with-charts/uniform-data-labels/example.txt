//ExStepSummary:0: एक नया खाली Word दस्तावेज़ बनाएं और उसे संपादित करने के लिए DocumentBuilder का उपयोग करें।
//ExStepSummary:1: दस्तावेज़ में 500×300 आकार का कॉलम चार्ट डालें और उसका Chart ऑब्जेक्ट प्राप्त करें।
//ExStepSummary:2: Aspose.Words द्वारा नए चार्ट में जोड़ी गई डिफ़ॉल्ट सैंपल सीरीज़ को हटाएँ।
//ExStepSummary:3: श्रेणी लेबल निर्धारित करें और दो डेटा सीरीज़ (Item Alpha और Item Beta) उनके मानों के साथ चार्ट में जोड़ें।
//ExStepSummary:4: बेसलाइन फ़ॉन्ट आकार और रंग सेट करें जो सभी डेटा लेबल पर लागू होगा।
//ExStepSummary:5: प्रत्येक सीरीज़ के लिए डेटा लेबल सक्षम करें, संख्यात्मक मान दिखाएँ, और पूर्वनिर्धारित फ़ॉन्ट आकार और रंग लागू करें।
//ExStepSummary:6: दस्तावेज़ को .docx फ़ाइल के रूप में सहेजें।
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document document = new Document();
DocumentBuilder builder = new DocumentBuilder(document);

//ExStep:1-
Shape chartContainer = builder.InsertChart(ChartType.Column, 500, 300);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();

//ExStep:3-
string[] periods = { "Period 1", "Period 2", "Period 3", "Period 4" };
chart.Series.Add("Item Alpha", periods, new double[] { 120, 150, 170, 130 });
chart.Series.Add("Item Beta", periods, new double[] { 80, 110, 140, 100 });

//ExStep:4-
const float labelFontSize = 10f;
Color labelFontColor = Color.White;

//ExStep:5-
foreach (ChartSeries series in chart.Series)
{
    series.HasDataLabels = true;
    series.DataLabels.ShowValue = true;
    series.DataLabels.Font.Size = labelFontSize;
    series.DataLabels.Font.Color = labelFontColor;
}

//ExStep:6-
document.Save("UniformLabelChart.docx");
