//ExSummary: यह ट्यूटोरियल दिखाता है कि कैसे एक नया Word दस्तावेज़ बनाएं, DocumentBuilder.InsertChart के साथ एक डिफ़ॉल्ट कॉलम चार्ट डालें, एम्बेडेड Chart ऑब्जेक्ट प्राप्त करें, और दस्तावेज़ को .docx फ़ाइल के रूप में सहेजें।
//ExStepSummary:0: एक खाली Word दस्तावेज़ बनाएं और उसे संपादित करने के लिए एक DocumentBuilder बनाएं।
//ExStepSummary:1: दस्तावेज़ में 432 × 252 पॉइंट्स आकार का एक कॉलम चार्ट डालें।
//ExStepSummary:2: यदि आवश्यक हो तो आगे की कस्टमाइज़ेशन के लिए डाले गए शेप से Chart ऑब्जेक्ट प्राप्त करें।
//ExStepSummary:3: दस्तावेज़ को, जिसमें अब डाला गया कॉलम चार्ट शामिल है, .docx फ़ाइल के रूप में सहेजें।
//ExStepImage:1:images/1.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);

//ExStep:2-
Chart embeddedChart = chartContainer.Chart;

//ExStep:3-
wordDoc.Save("column-demo.docx");
//ExEnd
