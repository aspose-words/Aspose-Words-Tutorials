//ExStepSummary:0: एक नया Word दस्तावेज़ बनाएं और उसे संपादित करने के लिए DocumentBuilder का उपयोग करें।
//ExStepSummary:1: दस्तावेज़ में 432×252 आकार का कॉलम चार्ट डालें और उसका Chart ऑब्जेक्ट प्राप्त करें।
//ExStepSummary:2: डिफ़ॉल्ट सीरीज़ को हटाएँ और तीन समूहों व मानों के साथ एक कस्टम सीरीज़ जोड़ें।
//ExStepSummary:3: चार्ट के प्राथमिक क्षैतिज (X) अक्ष का रेफ़रेंस प्राप्त करें।
//ExStepSummary:4: प्राथमिक X अक्ष पर प्रमुख ग्रिडलाइन के प्रदर्शन को सक्षम करें।
//ExStepSummary:5: ग्रिडलाइन का रंग नीला सेट करें और उनकी मोटाई 2 पॉइंट रखें।
//ExStepSummary:6: दस्तावेज़ को .docx फ़ाइल में सहेजें।
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
