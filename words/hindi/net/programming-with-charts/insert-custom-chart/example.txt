//ExStepSummary:0: एक नया Document बनाएं और उससे जुड़ा एक DocumentBuilder बनाएं।

//ExStepSummary:2: चार्ट से स्वचालित रूप से उत्पन्न नमूना सीरीज़ को हटा दें।
//ExStepSummary:3: X‑axis श्रेणी लेबल और प्रत्येक सीरीज़ के लिए लेजेंड एंट्रीज़ को परिभाषित करें।
//ExStepSummary:4: एक दो‑आयामी एरे बनाएं जहाँ प्रत्येक पंक्ति एक सीरीज़ के मान रखती है।
//ExStepSummary:5: प्रत्येक सीरीज़ पर इटररेट करें, मैट्रिक्स से उसकी पंक्ति के मान निकालें, और संबंधित लेजेंड एंट्री तथा X‑axis लेबल के साथ सीरीज़ को चार्ट में जोड़ें।
//ExStepSummary:6: चार्ट शीर्षक टेक्स्ट को 'Half‑Yearly Revenue' सेट करें और इसे दृश्यमान बनाएं।
//ExStepSummary:7: कस्टम चार्ट वाले दस्तावेज़ को .docx फ़ाइल के रूप में सहेजें।

//ExStep:0-
Document myDocument = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDocument);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chartObj = chartContainer.Chart;

//ExStep:2-
chartObj.Series.Clear();

//ExStep:3-
string[] xLabels = { "Jan", "Feb", "Mar", "Apr" };
string[] legendEntries = { "Item Alpha", "Item Beta", "Item Gamma" };

//ExStep:4-
double[,] dataMatrix = {
    { 120.5, 135.0, 150.2, 165.3 }, // Item Alpha
    { 80.0,  95.5, 110.1, 130.0 }, // Item Beta
    { 60.3,  70.4,  85.6,  95.2 }   // Item Gamma
};

//ExStep:5-
for (int sIdx = 0; sIdx < legendEntries.Length; sIdx++)
{
    double[] rowValues = new double[xLabels.Length];
    for (int cIdx = 0; cIdx < xLabels.Length; cIdx++)
    {
        rowValues[cIdx] = dataMatrix[sIdx, cIdx];
    }

    chartObj.Series.Add(legendEntries[sIdx], xLabels, rowValues);
}

//ExStep:6-
chartObj.Title.Text = "Half‑Yearly Revenue";
chartObj.Title.Show = true;

//ExStep:7-
myDocument.Save("CustomChart.docx");
