//ExSummary: यह ट्यूटोरियल दिखाता है कि Aspose.Words का उपयोग करके लाइन चार्ट वाला Word दस्तावेज़ कैसे बनाया जाए, द्वितीयक X‑अक्ष जोड़ा जाए, उसकी डिस्प्ले यूनिट को हजारों पर सेट किया जाए, और अक्ष लेबल्स पर कस्टम न्यूमेरिक फ़ॉर्मेट लागू किया जाए।
//ExStepSummary:0: एक नया Document और DocumentBuilder बनाएं ताकि Word फ़ाइल बनाना शुरू किया जा सके।
//ExStepSummary:1: दस्तावेज़ में 500×300 आकार का लाइन चार्ट डालें और Chart ऑब्जेक्ट प्राप्त करें।
//ExStepSummary:2: Aspose.Words द्वारा स्वचालित रूप से जोड़ी गई डिफ़ॉल्ट डेमो सीरीज़ को हटाएँ।
//ExStepSummary:3: X‑अक्ष के लिए उपयोग किए जाने वाले कैटेगरी लेबल्स की एक एरे परिभाषित करें।
//ExStepSummary:4: लेबल्स और वैल्यूज़ का उपयोग करके \"Main Data\" नाम की प्राइमरी डेटा सीरीज़ जोड़ें।
//ExStepSummary:5: Line प्रकार का एक सेकेंडरी सीरीज़ ग्रुप बनाएं और उसे सेकेंडरी एक्सिस ग्रुप को असाइन करें।
//ExStepSummary:6: सेकेंडरी X‑अक्ष की डिस्प्ले यूनिट को हजारों पर सेट करें ताकि मान K में दिखें।
//ExStepSummary:7: सेकेंडरी X‑अक्ष के टिक लेबल्स पर कस्टम न्यूमेरिक फ़ॉर्मेट \"#,##0.0\" लागू करें।
//ExStepSummary:8: वैकल्पिक रूप से सेकेंडरी ग्रुप में एक सहायक डेटा सीरीज़ जोड़ें।
//ExStepSummary:9: दस्तावेज़ को .docx फ़ाइल के रूप में सहेजें।

//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
