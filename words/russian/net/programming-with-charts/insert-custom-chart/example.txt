//ExSummary: В этом руководстве показано, как создать документ Word, вставить столбчатую диаграмму, очистить её данные по умолчанию и заполнить её пользовательскими данными из двумерного массива — сопоставляя каждую строку с рядом и каждый столбец с категорией — затем добавить заголовок и сохранить файл.
//ExStepSummary:0: Создайте новый Document и DocumentBuilder, привязанный к нему.
//ExStepSummary:1: Вставьте в документ столбчатую диаграмму размером 432×252 пункта и получите её объект Chart.
//ExStepSummary:2: Удалите автоматически сгенерированный образец ряда из диаграммы.
//ExStepSummary:3: Определите подписи категорий оси X и элементы легенды для каждого ряда.
//ExStepSummary:4: Создайте двумерный массив, где каждая строка содержит значения для одного ряда.
//ExStepSummary:5: Пройдитесь по каждому ряду, извлеките его значения из матрицы и добавьте ряд в диаграмму с соответствующим элементом легенды и подписями оси X.
//ExStepSummary:6: Установите текст заголовка диаграммы как 'Half‑Yearly Revenue' и сделайте его видимым.
//ExStepSummary:7: Сохраните документ, содержащий пользовательскую диаграмму, в файл формата .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:5:images/3.png
//ExStepImage:6:images/4.png
//ExStep:0-
Document myDocument = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDocument);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chartObj = chartContainer.Chart;

//ExStep:2-
chartObj.Series.Clear();

//ExStep:3-
string[] xLabels = { "Jan", "Feb", "Mar", "Apr" };
string[] legendEntries = { "Item Alpha", "Item Beta", "Item Gamma" };

//ExStep:4-
double[,] dataMatrix = {
    { 120.5, 135.0, 150.2, 165.3 }, // Item Alpha
    { 80.0,  95.5, 110.1, 130.0 }, // Item Beta
    { 60.3,  70.4,  85.6,  95.2 }   // Item Gamma
};

//ExStep:5-
for (int sIdx = 0; sIdx < legendEntries.Length; sIdx++)
{
    double[] rowValues = new double[xLabels.Length];
    for (int cIdx = 0; cIdx < xLabels.Length; cIdx++)
    {
        rowValues[cIdx] = dataMatrix[sIdx, cIdx];
    }

    chartObj.Series.Add(legendEntries[sIdx], xLabels, rowValues);
}

//ExStep:6-
chartObj.Title.Text = "Half‑Yearly Revenue";
chartObj.Title.Show = true;

//ExStep:7-
myDocument.Save("CustomChart.docx");
//ExEnd
