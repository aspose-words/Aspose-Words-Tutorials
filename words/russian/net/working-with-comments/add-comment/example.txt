//ExSummary: Этот учебник показывает, как создать новый документ Word, добавить три абзаца, найти определённый абзац, прикрепить к нему комментарий с информацией об авторе и сохранить результат как файл DOCX с использованием Aspose.Words.
//ExStepSummary:0: Создайте новый объект Document и DocumentBuilder, чтобы начать построение файла Word.
//ExStepSummary:1: Используйте DocumentBuilder.Writeln, чтобы добавить в документ три отдельных абзаца текста.
//ExStepSummary:2: Безопасно получайте второй абзац, переходя к первому, если в документе менее двух абзацев.
//ExStepSummary:3: Создайте объект Comment с данными автора, задайте его текст и прикрепите к выбранному абзацу.
//ExStepSummary:4: Сохраните изменённый документ как файл DOCX с именем 'sample.docx' в текущей папке.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
