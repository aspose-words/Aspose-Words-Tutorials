//ExStepSummary:3: Вызовите `builder.InsertComboBox`, чтобы разместить комбо‑бокс с именем "DropDown" и заданными элементами в текущей позиции курсора.

string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
string[] items = { "One", "Two", "Three" };

//ExStep:2-

//ExStep:3-
