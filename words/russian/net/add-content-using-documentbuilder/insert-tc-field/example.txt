//ExStepSummary:0: Укажите путь к папке, в которой будет сохранён полученный документ.

//ExStepSummary:2: Вставьте поле TC с текстом записи "Entry Text", который можно использовать для оглавления.

//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
