//ExSummary: Этот учебный материал показывает, как автоматически добавить красный диагональный текстовый водяной знак к каждому новому документу Word, создаваемому в процессе пакетной генерации отчётов с использованием Aspose.Words Document и DocumentBuilder.
//ExStepSummary:0: Создайте папку "GeneratedReports", в которой будут сохраняться выходные файлы.
//ExStepSummary:1: Запустите цикл, который сгенерирует три отдельные документа.
//ExStepSummary:2: Создайте новый пустой объект документа Word.
//ExStepSummary:3: Используйте DocumentBuilder, чтобы записать строку заголовка и описание в документ.
//ExStepSummary:4: Определите внешний вид водяного знака, включая шрифт, размер, цвет и диагональное расположение.
//ExStepSummary:5: Примените настроенный красный диагональный водяной знак с текстом "PROTECTED" к документу.
//ExStepSummary:6: Сохраните документ с водяным знаком в папку "GeneratedReports" под уникальным именем файла.
//ExStepSummary:7: Закройте цикл после обработки текущего документа.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
