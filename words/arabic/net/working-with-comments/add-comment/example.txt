//ExSummary: يوضح هذا البرنامج التعليمي كيفية إنشاء مستند Word جديد، وإضافة ثلاثة فقرات، وتحديد فقرة معينة، وإرفاق تعليق يحتوي على معلومات المؤلف بها، وحفظ النتيجة كملف DOCX باستخدام Aspose.Words.
//ExStepSummary:0: إنشاء كائن Document جديد وDocumentBuilder لبدء بناء ملف Word.
//ExStepSummary:1: استخدم DocumentBuilder.Writeln لإضافة ثلاثة فقرات نصية منفصلة إلى المستند.
//ExStepSummary:2: استرجع الفقرة الثانية بأمان، مع الرجوع إلى الأولى إذا كان المستند يحتوي على أقل من فقرتين.
//ExStepSummary:3: أنشئ Comment مع تفاصيل المؤلف، حدّد نصه، وأرفقه بالفقرة المختارة.
//ExStepSummary:4: احفظ المستند المعدل كملف DOCX باسم 'sample.docx' في المجلد الحالي.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
