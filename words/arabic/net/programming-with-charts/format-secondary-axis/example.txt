//ExSummary: يوضح هذا البرنامج التعليمي كيفية إنشاء مستند Word يحتوي على مخطط خطي، وإضافة محور X ثانوي، وتعيين وحدة العرض إلى الآلاف، وتطبيق تنسيق رقمي مخصص على تسميات المحور باستخدام Aspose.Words.
//ExStepSummary:0: أنشئ كائن Document جديد و DocumentBuilder لبدء بناء ملف Word.
//ExStepSummary:1: أدرج مخططًا خطيًا بحجم 500×300 في المستند واحصل على كائن Chart.
//ExStepSummary:2: احذف سلسلة العرض التجريبية الافتراضية التي يضيفها Aspose.Words تلقائيًا.
//ExStepSummary:3: عرّف مصفوفة من تسميات الفئات التي ستُستخدم لمحور X.
//ExStepSummary:4: أضف سلسلة البيانات الأساسية المسماة \"Main Data\" باستخدام التسميات والقيم.
//ExStepSummary:5: أنشئ مجموعة سلسلة ثانوية من النوع Line وعيّنها إلى مجموعة المحور الثانوي.
//ExStepSummary:6: عيّن وحدة عرض محور X الثانوي إلى الآلاف بحيث تُعرض القيم بـ K.
//ExStepSummary:7: طبق تنسيق رقمي مخصص \"#,##0.0\" على تسميات علامات محور X الثانوي.
//ExStepSummary:8: يمكنك إضافة سلسلة بيانات مساعدة إلى المجموعة الثانوية اختياريًا.
//ExStepSummary:9: احفظ المستند كملف .docx.

//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
