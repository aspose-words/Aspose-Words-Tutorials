//ExSummary: Deze tutorial laat zien hoe je een Word‑document maakt, een kolomgrafiek invoegt, de voorbeeldgegevens vervangt, en vervolgens de hoofd‑roosterlijnen op de primaire X‑as inschakelt en opmaakt.
//ExStepSummary:0: Maak een nieuw Word‑document en een DocumentBuilder om het te bewerken.
//ExStepSummary:1: Voeg een 432×252 kolomgrafiek toe aan het document en verkrijg het Chart‑object.
//ExStepSummary:2: Verwijder de standaard‑reeks en voeg een aangepaste reeks toe met drie groepen en waarden.
//ExStepSummary:3: Krijg een referentie naar de primaire horizontale (X‑)as van de grafiek.
//ExStepSummary:4: Schakel de weergave van hoofd‑roosterlijnen in op de primaire X‑as.
//ExStepSummary:5: Stel de kleur van de roosterlijnen in op blauw en hun dikte op 2 punten.
//ExStepSummary:6: Sla het document op als een .docx‑bestand.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
