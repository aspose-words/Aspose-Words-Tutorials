//ExSummary: Deze tutorial laat zien hoe je een Word‑document maakt met een lijndiagram, een secundaire X‑as toevoegt, de weergave‑eenheid instelt op duizenden, en een aangepast numeriek formaat toepast op de as‑labels met behulp van Aspose.Words.
//ExStepSummary:0: Doe een nieuw Document en een DocumentBuilder aan om te beginnen met het bouwen van het Word‑bestand.
//ExStepSummary:1: Voeg een lijndiagram van grootte 500×300 in het document in en verkrijg het Chart‑object.
//ExStepSummary:2: Verwijder de standaard demo‑reeks die Aspose.Words automatisch toevoegt.
//ExStepSummary:3: Definieer een array met categorielabels die voor de X‑as worden gebruikt.
//ExStepSummary:4: Voeg de primaire gegevensreeks met de naam \"Main Data\" toe met behulp van de labels en waarden.
//ExStepSummary:5: Maak een secundaire seriesgroep van het type Line aan en wijs deze toe aan de secundaire as‑groep.
//ExStepSummary:6: Stel de weergave‑eenheid van de secundaire X‑as in op duizenden zodat waarden worden weergegeven als K.
//ExStepSummary:7: Pas een aangepast numeriek formaat \"#,##0.0\" toe op de tick‑labels van de secundaire X‑as.
//ExStepSummary:8: Voeg optioneel een aanvullende gegevensreeks toe aan de secundaire groep.
//ExStepSummary:9: Sla het document op als een .docx‑bestand.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStepImage:6:images/5.png
//ExStepImage:7:images/6.png
//ExStepImage:8:images/7.png
//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
