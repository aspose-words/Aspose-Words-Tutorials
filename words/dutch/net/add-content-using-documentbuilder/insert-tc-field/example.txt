//ExStepSummary:0: Definieer het mappad waar het resulterende document wordt opgeslagen.
//ExStepSummary:1: Maak een nieuw leeg Document en een DocumentBuilder die eraan gekoppeld is voor bewerking.
//ExStepSummary:2: Voeg een TC‑veld in met invoertekst "Entry Text" dat kan worden gebruikt voor een inhoudsopgave.
//ExStepSummary:3: Sla het gewijzigde document op in de opgegeven map.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
