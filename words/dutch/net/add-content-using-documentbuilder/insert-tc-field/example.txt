//ExSummary: Deze tutorial laat zien hoe je een nieuw Word‑document maakt, DocumentBuilder gebruikt om een TC‑veld (inhoudsopgave) in te voegen, en het resulterende bestand opslaat.
//ExStepSummary:0: Definieer het mappad waar het gegenereerde document wordt opgeslagen.
//ExStepSummary:1: Instantieer een leeg Document en een DocumentBuilder die het bewerkt.
//ExStepSummary:2: Voeg een TC‑veld toe met de invoertekst \"Entry Text\" die zal worden gebruikt voor de inhoudsopgave.
//ExStepSummary:3: Sla het gewijzigde document op in de opgegeven map.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
