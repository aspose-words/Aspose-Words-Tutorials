//ExSummary: Αυτό το tutorial δείχνει πώς να δημιουργήσετε ένα νέο έγγραφο Word, να προσθέσετε τρία παραγράφους, να εντοπίσετε μια συγκεκριμένη παράγραφο, να επισυνάψετε ένα σχόλιο με πληροφορίες συγγραφέα σε αυτήν, και να αποθηκεύσετε το αποτέλεσμα ως αρχείο DOCX χρησιμοποιώντας το Aspose.Words.
//ExStepSummary:0: Δημιουργήστε ένα νέο Document και ένα DocumentBuilder για να ξεκινήσετε τη δημιουργία του αρχείου Word.
//ExStepSummary:1: Χρησιμοποιήστε το DocumentBuilder.Writeln για να προσθέσετε τρεις ξεχωριστές παραγράφους κειμένου στο έγγραφο.
//ExStepSummary:2: Ανακτήστε τη δεύτερη παράγραφο με ασφάλεια, επιστρέφοντας στην πρώτη εάν το έγγραφο έχει λιγότερες από δύο παραγράφους.
//ExStepSummary:3: Δημιουργήστε ένα Comment με τα στοιχεία του συγγραφέα, ορίστε το κείμενό του και επισυνάψτε το στην επιλεγμένη παράγραφο.
//ExStepSummary:4: Αποθηκεύστε το τροποποιημένο έγγραφο ως αρχείο DOCX με όνομα 'sample.docx' στον τρέχοντα φάκελο.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
