//ExSummary: Χρησιμοποιώντας το Aspose.Words, αυτό το σεμινάριο δημιουργεί ένα έγγραφο Word, εισάγει ένα διάγραμμα στήλης, ενημερώνει τον τίτλο του διαγράμματος με κείμενο που παρέχεται από τον χρήστη και εναλλάσσει την ορατότητα και την επικάλυψη του υπομνήματος βάσει των προτιμήσεων του χρήστη.
//ExStepSummary:0: Ορίστε τον τίτλο του διαγράμματος που παρέχεται από τον χρήστη και αν πρέπει να εμφανίζεται το υπόμνημα.
//ExStepSummary:1: Δημιουργήστε ένα νέο έγγραφο Word και ένα DocumentBuilder για να προσθέσετε περιεχόμενο.
//ExStepSummary:2: Εισάγετε ένα διάγραμμα στήλης 432 × 252 pt στο έγγραφο και επαληθεύστε ότι το σχήμα περιέχει πραγματικά ένα διάγραμμα.
//ExStepSummary:3: Ανακτήστε το αντικείμενο chart, ορίστε το κείμενο του τίτλου του στην τιμή που δόθηκε από τον χρήστη και εξασφαλίστε ότι ο τίτλος εμφανίζεται.
//ExStepSummary:4: Διαμορφώστε το υπόμνημα: εμφανίστε το στα δεξιά αν ζητηθεί, διαφορετικά κρύψτε το, και επιτρέψτε του να επικαλύπτει την περιοχή σχεδίασης.
//ExStepSummary:5: Αποθηκεύστε το τροποποιημένο έγγραφο στο δίσκο ως CustomChart.docx.
//ExStepImage:2:images/1.png
//ExStepImage:3:images/2.png
//ExStepImage:4:images/3.png
//ExStep:0-
string chartHeader = "Fiscal Q1 Performance Summary";
bool displayLegend = false;

//ExStep:1-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:2-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
if (!chartContainer.HasChart)
    throw new InvalidOperationException("The inserted shape does not contain a chart.");

//ExStep:3-
Chart chartObj = chartContainer.Chart;
ChartTitle chartTitle = chartObj.Title;
chartTitle.Text = chartHeader;
chartTitle.Show = true;

//ExStep:4-
ChartLegend chartLegend = chartObj.Legend;
chartLegend.Position = displayLegend ? LegendPosition.Right : LegendPosition.None;
chartLegend.Overlay = true;

//ExStep:5-
wordDoc.Save("CustomChart.docx");
//ExEnd
