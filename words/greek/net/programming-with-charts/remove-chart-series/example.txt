//ExSummary: Χρησιμοποιώντας το Aspose.Words για .NET, αυτό το σεμινάριο δημιουργεί ένα Document, εισάγει ένα γράφημα στήλης μέσω του DocumentBuilder, αφαιρεί μια συγκεκριμένη σειρά δεδομένων από το γράφημα με τη μέθοδο ChartSeriesCollection.RemoveAt και αποθηκεύει το τροποποιημένο έγγραφο.
//ExStepSummary:0: Δημιουργήστε ένα νέο Word Document και έναν DocumentBuilder που είναι συνδεδεμένος με αυτό.
//ExStepSummary:1: Εισάγετε ένα γράφημα στήλης μεγέθους 432×252 σημεία και αποκτήστε το αντικείμενο Chart για περαιτέρω επεξεργασία.
//ExStepSummary:2: Ορίστε το μηδενικό (zero‑based) δείκτη της σειράς που πρέπει να αφαιρεθεί (η δεύτερη σειρά).
//ExStepSummary:3: Αν ο δείκτης είναι έγκυρος, καλέστε τη μέθοδο RemoveAt στη συλλογή Series του γραφήματος για να διαγράψετε αυτή τη σειρά.
//ExStepSummary:4: Αποθηκεύστε το ενημερωμένο έγγραφο σε αρχείο .docx.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart columnChart = chartContainer.Chart;

//ExStep:2-
int targetSeriesIdx = 1;

//ExStep:3-
if (targetSeriesIdx >= 0 && targetSeriesIdx < columnChart.Series.Count)
{
    columnChart.Series.RemoveAt(targetSeriesIdx);
}

//ExStep:4-
wordDoc.Save("ChartWithoutSeries.docx");
//ExEnd
