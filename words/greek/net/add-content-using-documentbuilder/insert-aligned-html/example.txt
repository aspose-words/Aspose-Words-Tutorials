//ExSummary: Χρησιμοποιώντας το Aspose.Words Document και το DocumentBuilder, αυτό το σεμινάριο δείχνει πώς να εισάγετε ακατέργαστο περιεχόμενο HTML σε ένα έγγραφο Word.
//ExStepSummary:0: Ορίστε τη διαδρομή του φακέλου όπου θα αποθηκευτεί το τελικό έγγραφο.
//ExStepSummary:1: Δημιουργήστε μια νέα, κενή παρουσία εγγράφου Word.
//ExStepSummary:2: Δημιουργήστε ένα αντικείμενο DocumentBuilder που συνδέεται με το νεοδημιουργημένο έγγραφο.
//ExStepSummary:3: Καλέστε το `builder.InsertHtml()` για να ενσωματώσετε το παρεχόμενο HTML markup ως μορφοποιημένο περιεχόμενο Word.
//ExStepSummary:4: Αποθηκεύστε το τροποποιημένο έγγραφο στον καθορισμένο κατάλογο.
//ExStepImage:3:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();

//ExStep:2-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:3-
builder.InsertHtml(
    "<P align='right'>Paragraph right</P>" +
    "<b>Implicit paragraph left</b>" +
    "<div align='center'>Div center</div>" +
    "<h1 align='left'>Heading 1 left.</h1>");

//ExStep:4-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHtml.docx");
