//ExSummary: Bu öğreticide, yeni bir Word belgesi oluşturma, üç paragraf ekleme, belirli bir paragrafı bulma, ona yazar bilgileri içeren bir yorum ekleme ve sonucu Aspose.Words kullanarak DOCX dosyası olarak kaydetme gösterilmektedir.
//ExStepSummary:0: Word dosyasını oluşturmaya başlamak için yeni bir Document ve bir DocumentBuilder örneği oluşturun.
//ExStepSummary:1: Belgeye üç ayrı metin paragrafı eklemek için DocumentBuilder.Writeln kullanın.
//ExStepSummary:2: Belgenin iki paragraftan az olması durumunda, ikinci paragrafı güvenli bir şekilde al, aksi takdirde ilk paragrafı kullan.
//ExStepSummary:3: Yazar detaylarıyla bir Comment oluştur, metnini ayarla ve seçilen paragrafa ekle.
//ExStepSummary:4: Değiştirilen belgeyi geçerli klasörde 'sample.docx' adıyla bir DOCX dosyası olarak kaydedin.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
