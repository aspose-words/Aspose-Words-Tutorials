//ExSummary: Aspose.Words Document ve DocumentBuilder kullanarak, bu öğreticide ham HTML içeriğinin bir Word belgesine nasıl ekleneceği gösterilmektedir.
//ExStepSummary:0: Oluşturulan belgenin kaydedileceği klasör yolunu tanımlayın.
//ExStepSummary:1: Yeni, boş bir Word belge örneği oluşturun.
//ExStepSummary:2: Yeni oluşturulan belgeye bağlı bir DocumentBuilder örneği oluşturun.
//ExStepSummary:3: `builder.InsertHtml()` metodunu çağırarak sağlanan HTML işaretlemesini biçimlendirilmiş Word içeriği olarak gömün.
//ExStepSummary:4: Değiştirilen belgeyi belirtilen dizine kaydedin.
//ExStepImage:3:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();

//ExStep:2-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:3-
builder.InsertHtml(
    "<P align='right'>Paragraph right</P>" +
    "<b>Implicit paragraph left</b>" +
    "<div align='center'>Div center</div>" +
    "<h1 align='left'>Heading 1 left.</h1>");

//ExStep:4-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHtml.docx");
//ExEnd
