//ExSummary: Bu öğretici, Aspose.Words Document ve DocumentBuilder kullanarak toplu rapor sürecinde oluşturulan her yeni Word belgesine otomatik olarak kırmızı çapraz metin filigranı eklemeyi gösterir.
//ExStepSummary:0: "GeneratedReports" klasörünü oluşturun; çıktı dosyaları bu klasöre kaydedilecektir.
//ExStepSummary:1: Üç ayrı belge oluşturacak bir döngü başlatın.
//ExStepSummary:2: Yeni boş bir Word belge nesnesi oluşturun.
//ExStepSummary:3: Belgeye bir başlık satırı ve açıklama eklemek için DocumentBuilder'ı kullanın.
//ExStepSummary:4: Filigranın görünümünü, yazı tipi, boyut, renk ve çapraz düzen dahil olmak üzere tanımlayın.
//ExStepSummary:5: Belgeye "PROTECTED" metniyle yapılandırılmış kırmızı çapraz filigranı uygulayın.
//ExStepSummary:6: Filigranlı belgeyi benzersiz bir dosya adıyla "GeneratedReports" klasörüne kaydedin.
//ExStepSummary:7: Mevcut belge işlendikten sonra döngüyü kapatın.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
