//ExSummary: Bu öğreticide yeni bir Word belgesi oluşturma, bir DISPLAYBARCODE alanı ekleme, bunu DataMatrix barkodu için yapılandırma ve barkod görüntüsünü Document ve DocumentBuilder sınıflarını kullanarak oluşturma gösterilmektedir.
//ExStepSummary:0: Yeni boş bir Word Document oluşturun ve onu düzenlemek için bir DocumentBuilder kullanın.
//ExStepSummary:1: Mevcut imleç konumuna bir DISPLAYBARCODE alanı ekleyin; bu, belgeye bir alan yer tutucusu ekler.
//ExStepSummary:2: Alanının BarcodeType özelliğini DataMatrix olarak ayarlayın ve kodlanacak veri dizesini sağlayın.
//ExStepSummary:3: İsteğe bağlı olarak barkodun arka plan ve ön plan renklerini tanımlayın.
//ExStepSummary:4: Belge üzerinde UpdateFields metodunu çağırarak alan içinde barkod görüntüsünü oluşturun.
//ExStepSummary:5: Belgeyi bir .docx dosyasına kaydedin.
//ExStepImage:4:images/1.png
//ExStep:0-
// Initialise an empty Word document.
Document document = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
// Add a DISPLAYBARCODE field to the document.
FieldDisplayBarcode displayBarcodeField = (FieldDisplayBarcode)docBuilder.InsertField(FieldType.FieldDisplayBarcode, true);

//ExStep:2-
// Define the barcode format as DataMatrix and assign a sample payload.
displayBarcodeField.BarcodeType = "DataMatrix";
displayBarcodeField.BarcodeValue = "9876543210";

//ExStep:3-
// (Optional) Specify background and foreground colours.
displayBarcodeField.BackgroundColor = "0xFFFFFF";
displayBarcodeField.ForegroundColor = "0x000000";

//ExStep:4-
// Refresh all fields so the barcode image is rendered.
document.UpdateFields();

//ExStep:5-
// Persist the document to a file.
document.Save("DataMatrix_BarcodeDemo.docx");
//ExEnd
