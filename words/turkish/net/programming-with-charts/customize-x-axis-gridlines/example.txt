//ExSummary: Bu öğreticide, bir Word belgesi oluşturma, bir sütun grafiği ekleme, örnek verilerini değiştirme ve ardından birincil X eksenindeki ana ızgara çizgilerini etkinleştirip biçimlendirme gösterilmektedir.
//ExStepSummary:0: Yeni bir Word belgesi ve onu düzenlemek için bir DocumentBuilder oluşturun.
//ExStepSummary:1: Belgeye 432×252 boyutlarında bir sütun grafiği ekleyin ve ilgili Chart nesnesini alın.
//ExStepSummary:2: Varsayılan seriyi kaldırın ve üç grup ve değer içeren özel bir seri ekleyin.
//ExStepSummary:3: Grafiğin birincil yatay (X) eksenine bir referans alın.
//ExStepSummary:4: Birincil X ekseninde ana ızgara çizgilerinin görüntülenmesini etkinleştirin.
//ExStepSummary:5: Izgara çizgilerinin rengini maviye ve kalınlığını 2 puana ayarlayın.
//ExStepSummary:6: Belgeyi bir .docx dosyasına kaydedin.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
