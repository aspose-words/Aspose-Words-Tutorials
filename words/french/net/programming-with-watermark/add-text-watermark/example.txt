//ExSummary: Ce tutoriel montre comment ajouter automatiquement un filigrane texte rouge en diagonale à chaque nouveau document Word généré dans un processus de rapport par lots en utilisant Aspose.Words Document et DocumentBuilder.
//ExStepSummary:0: Créez le dossier "GeneratedReports" où les fichiers de sortie seront enregistrés.
//ExStepSummary:1: Démarrez une boucle qui générera trois documents distincts.
//ExStepSummary:2: Créez un nouvel objet document Word vide.
//ExStepSummary:3: Utilisez DocumentBuilder pour écrire une ligne de titre et une description dans le document.
//ExStepSummary:4: Définissez l'apparence du filigrane, y compris la police, la taille, la couleur et la disposition diagonale.
//ExStepSummary:5: Appliquez le filigrane rouge diagonal configuré avec le texte "PROTECTED" au document.
//ExStepSummary:6: Enregistrez le document filigrané dans le dossier "GeneratedReports" avec un nom de fichier unique.
//ExStepSummary:7: Fermez la boucle après le traitement du document actuel.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
