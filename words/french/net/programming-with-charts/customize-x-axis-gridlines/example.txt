//ExSummary: Ce tutoriel montre comment créer un document Word, insérer un graphique en colonnes, remplacer ses données d'exemple, puis activer et styliser les lignes de grille majeures sur l'axe X principal.
//ExStepSummary:0: Créez un nouveau document Word et un DocumentBuilder pour le modifier.
//ExStepSummary:1: Insérez un graphique en colonnes de 432×252 dans le document et obtenez son objet Chart.
//ExStepSummary:2: Supprimez la série par défaut et ajoutez une série personnalisée avec trois groupes et valeurs.
//ExStepSummary:3: Obtenez une référence à l'axe horizontal (X) principal du graphique.
//ExStepSummary:4: Activez l'affichage des lignes de grille majeures sur l'axe X principal.
//ExStepSummary:5: Définissez la couleur des lignes de grille en bleu et leur épaisseur à 2 points.
//ExStepSummary:6: Enregistrez le document dans un fichier .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
