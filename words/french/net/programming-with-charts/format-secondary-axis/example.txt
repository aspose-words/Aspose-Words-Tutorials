//ExStepSummary:0: Créez un nouveau document Word vide et un DocumentBuilder pour le modifier.
//ExStepSummary:1: Insérez un graphique en courbes de taille 450×250 points dans le document et récupérez son objet Chart.
//ExStepSummary:2: Supprimez la série de remplacement par défaut qu’Aspose ajoute à un nouveau graphique.
//ExStepSummary:3: Définissez les libellés des trimestres et ajoutez une série de données principale avec des valeurs d'exemple au graphique.
//ExStepSummary:4: Créez un groupe de séries secondaire de type Line et affectez-le au groupe d'axes secondaire.
//ExStepSummary:5: Masquez l'axe X secondaire afin que seul l'axe Y secondaire soit visible.
//ExStepSummary:6: Affichez le titre de l'axe Y secondaire et définissez son texte sur \"Sales (USD)\".
//ExStepSummary:7: Définissez le format numérique de l'axe Y secondaire sur un modèle de devise avec deux décimales et dissociez-le du formatage source.
//ExStepSummary:8: Ajoutez une série auxiliaire au groupe secondaire, en fournissant des données qui seront tracées par rapport à l'axe Y secondaire.
//ExStepSummary:9: Enregistrez le document final dans un fichier nommé ProfitChart.docx dans le répertoire actuel.

//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape lineChartShape = docBuilder.InsertChart(ChartType.Line, 450, 250);
Chart lineChart = lineChartShape.Chart;

//ExStep:2-
lineChart.Series.Clear();

//ExStep:3-
string[] quarters = new[] { "Q1", "Q2", "Q3", "Q4" };
lineChart.Series.Add("Main Series", quarters, new double[] { 1200, 1500, 1800, 2100 });

//ExStep:4-
ChartSeriesGroup secondarySeriesGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
secondarySeriesGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:5-
secondarySeriesGroup.AxisX.Hidden = true;

//ExStep:6-
secondarySeriesGroup.AxisY.Title.Show = true;
secondarySeriesGroup.AxisY.Title.Text = "Sales (USD)";

//ExStep:7-
secondarySeriesGroup.AxisY.NumberFormat.FormatCode = "\"$\"#,##0.00";
secondarySeriesGroup.AxisY.NumberFormat.IsLinkedToSource = false;

//ExStep:8-
secondarySeriesGroup.Series.Add("Auxiliary Series", quarters, new double[] { 3000, 3500, 4000, 4500 });

//ExStep:9-
string resultPath = Path.Combine(Directory.GetCurrentDirectory(), "ProfitChart.docx");
wordDoc.Save(resultPath);
