//ExSummary: Ce tutoriel montre comment créer un document Word avec un graphique en courbes, ajouter un axe X secondaire, définir son unité d’affichage en milliers et appliquer un format numérique personnalisé aux étiquettes d’axe en utilisant Aspose.Words.
//ExStepSummary:0: Créez un nouvel objet Document et un DocumentBuilder pour commencer à construire le fichier Word.
//ExStepSummary:1: Insérez un graphique en courbes de taille 500×300 dans le document et récupérez l’objet Chart.
//ExStepSummary:2: Supprimez la série de démonstration par défaut qu’Aspose.Words ajoute automatiquement.
//ExStepSummary:3: Définissez un tableau d’étiquettes de catégorie qui seront utilisées pour l’axe X.
//ExStepSummary:4: Ajoutez la série de données principale nommée "Main Data" en utilisant les étiquettes et les valeurs.
//ExStepSummary:5: Créez un groupe de séries secondaire de type Line et affectez-le au groupe d’axes secondaire.
//ExStepSummary:6: Définissez l’unité d’affichage de l’axe X secondaire sur des milliers afin que les valeurs soient affichées en K.
//ExStepSummary:7: Appliquez le format numérique personnalisé "#,##0.0" aux graduations de l’axe X secondaire.
//ExStepSummary:8: Ajoutez éventuellement une série de données auxiliaire au groupe secondaire.
//ExStepSummary:9: Enregistrez le document dans un fichier .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStepImage:6:images/5.png
//ExStepImage:7:images/6.png
//ExStepImage:8:images/7.png
//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
