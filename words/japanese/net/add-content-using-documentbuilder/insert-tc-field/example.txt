//ExSummary: Aspose.Words for .NET を使用して、このチュートリアルは新しい Word ドキュメントを作成し、DocumentBuilder を使ってカスタムエントリテキストを持つ TC（目次）フィールドを挿入し、ファイルを保存します。
//ExStepSummary:0: 生成されたドキュメントを保存するフォルダー パスを定義します。
//ExStepSummary:1: 新しい空の Document を作成し、編集用にそれに紐付けた DocumentBuilder を作成します。
//ExStepSummary:2: 目次に使用できるエントリテキスト "Entry Text" を持つ TC フィールドを挿入します。
//ExStepSummary:3: 変更されたドキュメントを指定されたディレクトリに保存します。
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
