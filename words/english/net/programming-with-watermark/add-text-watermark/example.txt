//ExStart
//ExSummary: This tutorial shows how to automatically add a red diagonal text watermark to every new Word document generated in a batch report process using Aspose.Words Document and DocumentBuilder.
//ExStepSummary:0: Create the "GeneratedReports" folder where the output files will be saved.
//ExStepSummary:1: Start a loop that will generate three separate documents.
//ExStepSummary:2: Create a new empty Word document object.
//ExStepSummary:3: Use DocumentBuilder to write a title line and a description into the document.
//ExStepSummary:4: Define the appearance of the watermark, including font, size, color, and diagonal layout.
//ExStepSummary:5: Apply the configured red diagonal watermark with the text "PROTECTED" to the document.
//ExStepSummary:6: Save the watermarked document to the "GeneratedReports" folder with a unique filename.
//ExStepSummary:7: Close the loop after processing the current document.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
