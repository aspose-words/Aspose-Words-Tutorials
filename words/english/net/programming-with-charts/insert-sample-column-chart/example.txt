//ExSummary: This tutorial demonstrates how to create a new Word document, insert a default column chart with DocumentBuilder.InsertChart, obtain the embedded Chart object, and save the document to a .docx file.
//ExStepSummary:0: Create an empty Word document and a DocumentBuilder to edit it.
//ExStepSummary:1: Insert a column chart of size 432 × 252 points into the document.
//ExStepSummary:2: Retrieve the Chart object from the inserted shape for further customization if needed.
//ExStepSummary:3: Save the document, which now contains the inserted column chart, to a .docx file.
//ExStepImage:1:images/1.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);

//ExStep:2-
Chart embeddedChart = chartContainer.Chart;

//ExStep:3-
wordDoc.Save("column-demo.docx");
//ExEnd
