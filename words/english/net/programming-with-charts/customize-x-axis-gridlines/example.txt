//ExSummary: This tutorial shows how to create a Word document, insert a column chart, replace its sample data, and then enable and style the major gridlines on the primary X axis.
//ExStepSummary:0: Create a new Word document and a DocumentBuilder to edit it.
//ExStepSummary:1: Insert a 432×252 column chart into the document and obtain its Chart object.
//ExStepSummary:2: Remove the default series and add a custom series with three groups and values.
//ExStepSummary:3: Get a reference to the chart's primary horizontal (X) axis.
//ExStepSummary:4: Enable the display of major gridlines on the primary X axis.
//ExStepSummary:5: Set the gridlines' color to blue and their weight to 2 points.
//ExStepSummary:6: Save the document to a .docx file.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
