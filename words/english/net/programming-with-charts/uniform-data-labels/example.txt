//ExStart
//ExSummary: This tutorial demonstrates how to create a Word document, insert a column chart, add custom data series, and define default data‑label font size and colour that are applied uniformly to all series in the chart.
//ExStepSummary:0: Create a new empty Word document and a DocumentBuilder to edit it.
//ExStepSummary:1: Insert a 500×300 column chart into the document and obtain its Chart object.
//ExStepSummary:2: Remove the default sample series that Aspose.Words adds to a new chart.
//ExStepSummary:3: Define category labels and add two data series (Item Alpha and Item Beta) with their values to the chart.
//ExStepSummary:4: Set baseline font size and colour that will be applied to all data labels.
//ExStepSummary:5: Enable data labels for each series, show the numeric value, and apply the predefined font size and colour.
//ExStepSummary:6: Save the document to a .docx file.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document document = new Document();
DocumentBuilder builder = new DocumentBuilder(document);

//ExStep:1-
Shape chartContainer = builder.InsertChart(ChartType.Column, 500, 300);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();

//ExStep:3-
string[] periods = { "Period 1", "Period 2", "Period 3", "Period 4" };
chart.Series.Add("Item Alpha", periods, new double[] { 120, 150, 170, 130 });
chart.Series.Add("Item Beta", periods, new double[] { 80, 110, 140, 100 });

//ExStep:4-
const float labelFontSize = 10f;
Color labelFontColor = Color.White;

//ExStep:5-
foreach (ChartSeries series in chart.Series)
{
    series.HasDataLabels = true;
    series.DataLabels.ShowValue = true;
    series.DataLabels.Font.Size = labelFontSize;
    series.DataLabels.Font.Color = labelFontColor;
}

//ExStep:6-
document.Save("UniformLabelChart.docx");
//ExEnd
