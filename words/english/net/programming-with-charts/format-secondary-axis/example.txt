//ExStart
//ExSummary: This tutorial demonstrates how to create a Word document with a line chart, add a secondary X‑axis, set its display unit to thousands, and apply a custom numeric format to the axis labels using Aspose.Words.
//ExStepSummary:0: Create a new Document and a DocumentBuilder to start building the Word file.
//ExStepSummary:1: Insert a line chart of size 500×300 into the document and obtain the Chart object.
//ExStepSummary:2: Remove the default demo series that Aspose.Words adds automatically.
//ExStepSummary:3: Define an array of category labels that will be used for the X‑axis.
//ExStepSummary:4: Add the primary data series named "Main Data" using the labels and values.
//ExStepSummary:5: Create a secondary series group of type Line and assign it to the secondary axis group.
//ExStepSummary:6: Set the secondary X‑axis display unit to thousands so values are shown in K.
//ExStepSummary:7: Apply a custom numeric format "#,##0.0" to the secondary X‑axis tick labels.
//ExStepSummary:8: Optionally add an auxiliary data series to the secondary group.
//ExStepSummary:9: Save the document to a .docx file.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStepImage:6:images/5.png
//ExStepImage:7:images/6.png
//ExStepImage:8:images/7.png
//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
