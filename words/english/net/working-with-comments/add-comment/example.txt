//ExSummary: This tutorial shows how to create a new Word document, add three paragraphs, locate a specific paragraph, attach a comment with author information to it, and save the result as a DOCX file using Aspose.Words.
//ExStepSummary:0: Instantiate a new Document and a DocumentBuilder to start building the Word file.
//ExStepSummary:1: Use DocumentBuilder.Writeln to add three separate paragraphs of text to the document.
//ExStepSummary:2: Retrieve the second paragraph safely, falling back to the first if the document has fewer than two paragraphs.
//ExStepSummary:3: Create a Comment with author details, set its text, and attach it to the selected paragraph.
//ExStepSummary:4: Save the modified document as a DOCX file named 'sample.docx' in the current folder.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
