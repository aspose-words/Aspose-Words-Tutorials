//ExSummary: Using Aspose.Words Document, DocumentBuilder, Comment, and Style classes, this tutorial creates comments in a Word file and applies a custom paragraph style to every comment paragraph so the comment text matches the corporate brand.
//ExStepSummary:0: Create a new empty Document and a DocumentBuilder to edit it.
//ExStepSummary:1: Insert two example sentences into the document.
//ExStepSummary:2: Create a comment by Carol, set its text, move to the start of the document, and attach the comment to the first paragraph.
//ExStepSummary:3: Create a comment by Dave, set its text, move to the end of the document, and attach the comment to the last paragraph.
//ExStepSummary:4: Define a custom paragraph style named "BrandComment" with Arial, 10 pt, dark blue, bold formatting to match the corporate brand.
//ExStepSummary:5: Collect all Comment nodes in the document into a list.
//ExStepSummary:6: Iterate over each comment and apply the BrandComment style to every paragraph inside the comment.
//ExStepSummary:7: Create a Results folder if needed and save the modified document as StyledCommentsDocument.docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStepImage:6:images/4.png
//ExStep:0-
Document document = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
docBuilder.Writeln("Introductory line for the file.");
docBuilder.Writeln("Follow‑up line for the file.");

//ExStep:2-
Comment remarkOne = new Comment(document, "Carol", "C", DateTime.Now);
remarkOne.SetText("Preliminary assessment note.");
docBuilder.MoveToDocumentStart();
docBuilder.CurrentParagraph.AppendChild(remarkOne);

//ExStep:3-
Comment remarkTwo = new Comment(document, "Dave", "D", DateTime.Now);
remarkTwo.SetText("Further clarification required.");
docBuilder.MoveToDocumentEnd();
docBuilder.CurrentParagraph.AppendChild(remarkTwo);

//ExStep:4-
Style brandCommentStyle = document.Styles.Add(StyleType.Paragraph, "BrandComment");
brandCommentStyle.Font.Name = "Arial";
brandCommentStyle.Font.Size = 10;
brandCommentStyle.Font.Color = Color.DarkBlue;
brandCommentStyle.Font.Bold = true;

//ExStep:5-
var allComments = document.GetChildNodes(NodeType.Comment, true)
                                 .OfType<Comment>()
                                 .ToList();

//ExStep:6-
foreach (Comment commentNode in allComments)
{
    if (commentNode.Paragraphs.Count == 0)
        commentNode.EnsureMinimum();

    foreach (Paragraph para in commentNode.Paragraphs)
    {
        para.ParagraphFormat.Style = brandCommentStyle;
    }
}

//ExStep:7-
string resultFolder = Path.Combine(Directory.GetCurrentDirectory(), "Results");
Directory.CreateDirectory(resultFolder);
string resultFile = Path.Combine(resultFolder, "StyledCommentsDocument.docx");
document.Save(resultFile);
//ExEnd
