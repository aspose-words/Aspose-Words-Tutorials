//ExStepSummary:1: Erstellen Sie ein neues leeres Document und einen DocumentBuilder, der damit verknüpft ist, zum Bearbeiten.
//ExStepSummary:2: Fügen Sie ein TC‑Feld mit dem Eintragstext "Entry Text" ein, das für ein Inhaltsverzeichnis verwendet werden kann.
//ExStepSummary:3: Speichern Sie das geänderte Dokument im angegebenen Verzeichnis.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
