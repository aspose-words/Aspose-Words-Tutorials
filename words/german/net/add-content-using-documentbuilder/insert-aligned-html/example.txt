//ExSummary: Mit Aspose.Words Document und DocumentBuilder fügt dieses Tutorial HTML-Markup in ein Word-Dokument ein und speichert das Ergebnis.
//ExStepSummary:0: Definieren Sie den Ausgabepfad, erstellen Sie ein neues Document und initialisieren Sie einen DocumentBuilder für dieses Document.
//ExStepSummary:1: Rufen Sie `builder.InsertHtml()` auf, um einen rechtsbündigen Absatz, fett formatierten Text, ein zentriertes div und eine linksbündige Überschrift in das Dokument einzufügen.
//ExStepSummary:2: Speichern Sie das modifizierte Dokument im angegebenen Ordner als .docx-Datei.
//ExStepImage:1:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

Document doc = new Document();

DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:1-
builder.InsertHtml(
    "<P align='right'>Paragraph right</P>" +
    "<b>Implicit paragraph left</b>" +
    "<div align='center'>Div center</div>" +
    "<h1 align='left'>Heading 1 left.</h1>");

//ExStep:2-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHtml.docx");
