//ExSummary: Dieses Tutorial zeigt, wie man automatisch ein rotes diagonales Textwasserzeichen zu jedem neu erzeugten Word-Dokument hinzufügt, das in einem Batch‑Berichtsprozess mit Aspose.Words Document und DocumentBuilder generiert wird.
//ExStepSummary:0: Erstellen Sie den Ordner "GeneratedReports", in dem die Ausgabedateien gespeichert werden.
//ExStepSummary:1: Starten Sie eine Schleife, die drei separate Dokumente erzeugt.
//ExStepSummary:2: Erstellen Sie ein neues leeres Word-Dokumentobjekt.
//ExStepSummary:3: Verwenden Sie DocumentBuilder, um eine Titelzeile und eine Beschreibung in das Dokument zu schreiben.
//ExStepSummary:4: Definieren Sie das Aussehen des Wasserzeichens, einschließlich Schriftart, Größe, Farbe und diagonaler Anordnung.
//ExStepSummary:5: Wenden Sie das konfigurierte rote diagonale Wasserzeichen mit dem Text "PROTECTED" auf das Dokument an.
//ExStepSummary:6: Speichern Sie das wassergezeichnete Dokument im Ordner "GeneratedReports" mit einem eindeutigen Dateinamen.
//ExStepSummary:7: Schließen Sie die Schleife nach der Verarbeitung des aktuellen Dokuments.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
