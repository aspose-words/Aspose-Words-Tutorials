//ExSummary: Dieses Tutorial zeigt, wie man ein Word‑Dokument mit einem Säulendiagramm erstellt, es mit bestimmten Datenreihen füllt, überprüft, dass das Diagramm die erwartete Anzahl von Reihen und Datenpunkten enthält, und anschließend das Dokument speichert.
//ExStepSummary:0: Definieren Sie die erforderliche Anzahl von Reihen und Datenpunkten, die das Diagramm enthalten muss.
//ExStepSummary:1: Erstellen Sie ein neues Word‑Document und einen DocumentBuilder, um Inhalte hinzuzufügen.
//ExStepSummary:2: Fügen Sie ein 400×300‑Säulendiagramm in das Dokument ein und erhalten Sie das zugehörige Chart‑Objekt.
//ExStepSummary:3: Entfernen Sie die Platzhalter‑Datenreihe, die Aspose.Words standardmäßig hinzufügt.
//ExStepSummary:4: Definieren Sie X‑Achsen‑Beschriftungen und fügen Sie dem Diagramm zwei Datenreihen (Alpha und Beta) mit jeweils drei Punkten hinzu.
//ExStepSummary:5: Überprüfen Sie, ob das Diagramm exakt die erforderliche Anzahl von Reihen enthält, und werfen Sie eine Ausnahme, falls dies nicht der Fall ist.
//ExStepSummary:6: Iterieren Sie über jede Reihe und prüfen Sie, ob sie die erwartete Anzahl von Datenpunkten hat; bei einer Abweichung wird eine Ausnahme ausgelöst.
//ExStepSummary:7: Speichern Sie das validierte Dokument als .docx‑Datei.
//ExStepImage:2:images/1.png
//ExStepImage:3:images/2.png
//ExStepImage:4:images/3.png
//ExStep:0-
// Desired numbers for checks.
const int requiredSeriesCount = 2;
const int requiredPointsPerSeries = 3;

//ExStep:1-
// Initialise a fresh document and its builder.
Document myDocument = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDocument);

//ExStep:2-
// Place a column‑type chart into the document.
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 400, 300);
Chart chartObj = chartContainer.Chart;

//ExStep:3-
// Discard the sample data automatically added by Aspose.Words.
chartObj.Series.Clear();

//ExStep:4-
// Set up X‑axis labels and add two data series with matching values.
string[] xLabels = { "Group A", "Group B", "Group C" };
chartObj.Series.Add("Data Set Alpha", xLabels, new double[] { 10, 20, 30 });
chartObj.Series.Add("Data Set Beta", xLabels, new double[] { 15, 25, 35 });

//ExStep:5-
// ----- Validation -----
// Confirm the chart holds the expected number of series.
if (chartObj.Series.Count != requiredSeriesCount)
    throw new InvalidOperationException(
        $"Chart should contain {requiredSeriesCount} series, but actually has {chartObj.Series.Count}.");

//ExStep:6-
// Confirm each series has the required amount of data points.
foreach (ChartSeries seriesItem in chartObj.Series)
{
    if (seriesItem.DataPoints.Count != requiredPointsPerSeries)
        throw new InvalidOperationException(
            $"Series '{seriesItem.Name}' should have {requiredPointsPerSeries} points, but contains {seriesItem.DataPoints.Count}.");
}

//ExStep:7-
// Persist the verified document.
myDocument.Save("checked-graph.docx");
//ExEnd
