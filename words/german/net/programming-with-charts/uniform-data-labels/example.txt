//ExSummary: Dieses Tutorial zeigt, wie man ein Word‑Dokument erstellt, ein Säulendiagramm einfügt, benutzerdefinierte Datenreihen hinzufügt und die Standard‑Schriftgröße und -Farbe für Datenbeschriftungen definiert, die einheitlich auf alle Reihen im Diagramm angewendet werden.
//ExStepSummary:0: Erstellen Sie ein neues leeres Word‑Dokument und einen DocumentBuilder, um es zu bearbeiten.
//ExStepSummary:1: Fügen Sie ein 500 × 300‑Pixel‑Säulendiagramm in das Dokument ein und erhalten Sie das zugehörige Chart‑Objekt.
//ExStepSummary:2: Entfernen Sie die standardmäßige Beispielreihe, die Aspose.Words zu einem neuen Diagramm hinzufügt.
//ExStepSummary:3: Definieren Sie Kategorienbeschriftungen und fügen Sie dem Diagramm zwei Datenreihen (Item Alpha und Item Beta) mit ihren Werten hinzu.
//ExStepSummary:4: Legen Sie die Grundschriftgröße und -farbe fest, die auf alle Datenbeschriftungen angewendet wird.
//ExStepSummary:5: Aktivieren Sie Datenbeschriftungen für jede Reihe, zeigen Sie den numerischen Wert an und wenden Sie die vordefinierte Schriftgröße und -farbe an.
//ExStepSummary:6: Speichern Sie das Dokument als .docx‑Datei.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document document = new Document();
DocumentBuilder builder = new DocumentBuilder(document);

//ExStep:1-
Shape chartContainer = builder.InsertChart(ChartType.Column, 500, 300);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();

//ExStep:3-
string[] periods = { "Period 1", "Period 2", "Period 3", "Period 4" };
chart.Series.Add("Item Alpha", periods, new double[] { 120, 150, 170, 130 });
chart.Series.Add("Item Beta", periods, new double[] { 80, 110, 140, 100 });

//ExStep:4-
const float labelFontSize = 10f;
Color labelFontColor = Color.White;

//ExStep:5-
foreach (ChartSeries series in chart.Series)
{
    series.HasDataLabels = true;
    series.DataLabels.ShowValue = true;
    series.DataLabels.Font.Size = labelFontSize;
    series.DataLabels.Font.Color = labelFontColor;
}

//ExStep:6-
document.Save("UniformLabelChart.docx");
//ExEnd
