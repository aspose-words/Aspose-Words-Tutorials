//ExSummary: Dieses Tutorial zeigt, wie man ein Word‑Dokument mit einem Liniendiagramm erstellt, eine sekundäre X‑Achse hinzufügt, deren Anzeigeeinheit auf Tausend setzt und ein benutzerdefiniertes Zahlenformat auf die Achsenbeschriftungen anwendet, wobei Aspose.Words verwendet wird.
//ExStepSummary:0: Erstellen Sie ein neues Document und einen DocumentBuilder, um mit dem Aufbau der Word‑Datei zu beginnen.
//ExStepSummary:1: Fügen Sie ein Liniendiagramm mit der Größe 500×300 in das Dokument ein und erhalten Sie das Chart‑Objekt.
//ExStepSummary:2: Entfernen Sie die standardmäßige Demo‑Serie, die Aspose.Words automatisch hinzufügt.
//ExStepSummary:3: Definieren Sie ein Array von Kategorienbeschriftungen, das für die X‑Achse verwendet wird.
//ExStepSummary:4: Fügen Sie die primäre Datenreihe mit dem Namen "Main Data" unter Verwendung der Beschriftungen und Werte hinzu.
//ExStepSummary:5: Erstellen Sie eine sekundäre Series‑Group vom Typ Line und weisen Sie sie der sekundären Achsen‑Group zu.
//ExStepSummary:6: Setzen Sie die Anzeigeeinheit der sekundären X‑Achse auf Tausend, sodass Werte in K angezeigt werden.
//ExStepSummary:7: Wenden Sie das benutzerdefinierte Zahlenformat "#,##0.0" auf die Tick‑Beschriftungen der sekundären X‑Achse an.
//ExStepSummary:8: Fügen Sie optional eine Hilfsdatenreihe zur sekundären Gruppe hinzu.
//ExStepSummary:9: Speichern Sie das Dokument als .docx‑Datei.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStepImage:6:images/5.png
//ExStepImage:7:images/6.png
//ExStepImage:8:images/7.png
//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
