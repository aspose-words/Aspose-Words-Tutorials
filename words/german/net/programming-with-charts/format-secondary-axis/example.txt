//ExStepSummary:0: Erstellen Sie ein neues leeres Word‑Dokument und einen DocumentBuilder, um es zu bearbeiten.
//ExStepSummary:1: Fügen Sie ein Liniendiagramm mit der Größe 450 × 250 Punkte in das Dokument ein und erhalten Sie das zugehörige Chart‑Objekt.
//ExStepSummary:2: Entfernen Sie die standardmäßige Platzhalter‑Serie, die Aspose einem neuen Diagramm hinzufügt.
//ExStepSummary:3: Definieren Sie Quartals‑Beschriftungen und fügen Sie dem Diagramm eine primäre Datenreihe mit Beispielwerten hinzu.
//ExStepSummary:4: Erstellen Sie eine sekundäre Seriengruppe vom Typ Line und weisen Sie sie der sekundären Achsengruppe zu.
//ExStepSummary:5: Blenden Sie die sekundäre X‑Achse aus, sodass nur die sekundäre Y‑Achse sichtbar ist.
//ExStepSummary:6: Zeigen Sie den Titel der sekundären Y‑Achse an und setzen Sie den Text auf "Sales (USD)".
//ExStepSummary:7: Setzen Sie das Zahlenformat der sekundären Y‑Achse auf ein Währungsmuster mit zwei Dezimalstellen und lösen Sie die Verknüpfung mit der Quellformatierung.
//ExStepSummary:8: Fügen Sie der sekundären Gruppe eine Hilfsserie hinzu, die Daten liefert, die gegen die sekundäre Y‑Achse geplottet werden.
//ExStepSummary:9: Speichern Sie das fertige Dokument unter dem Dateinamen ProfitChart.docx im aktuellen Verzeichnis.

//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape lineChartShape = docBuilder.InsertChart(ChartType.Line, 450, 250);
Chart lineChart = lineChartShape.Chart;

//ExStep:2-
lineChart.Series.Clear();

//ExStep:3-
string[] quarters = new[] { "Q1", "Q2", "Q3", "Q4" };
lineChart.Series.Add("Main Series", quarters, new double[] { 1200, 1500, 1800, 2100 });

//ExStep:4-
ChartSeriesGroup secondarySeriesGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
secondarySeriesGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:5-
secondarySeriesGroup.AxisX.Hidden = true;

//ExStep:6-
secondarySeriesGroup.AxisY.Title.Show = true;
secondarySeriesGroup.AxisY.Title.Text = "Sales (USD)";

//ExStep:7-
secondarySeriesGroup.AxisY.NumberFormat.FormatCode = "\"$\"#,##0.00";
secondarySeriesGroup.AxisY.NumberFormat.IsLinkedToSource = false;

//ExStep:8-
secondarySeriesGroup.Series.Add("Auxiliary Series", quarters, new double[] { 3000, 3500, 4000, 4500 });

//ExStep:9-
string resultPath = Path.Combine(Directory.GetCurrentDirectory(), "ProfitChart.docx");
wordDoc.Save(resultPath);
