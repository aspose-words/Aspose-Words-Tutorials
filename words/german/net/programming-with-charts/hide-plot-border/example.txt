//ExSummary: Dieses Tutorial zeigt, wie man ein Word‑Dokument erstellt, ein Säulendiagramm einfügt, es mit Beispieldaten füllt, die Begrenzung des Diagrammbereichs ausblendet, während die Achsenlinien sichtbar bleiben, und das Dokument schließlich speichert.
//ExStepSummary:0: Erstellen Sie ein neues leeres Word‑Dokument und einen DocumentBuilder, um es zu bearbeiten.
//ExStepSummary:1: Fügen Sie ein 432×252‑Säulendiagramm in das Dokument ein und erhalten Sie das zugehörige Chart‑Objekt.
//ExStepSummary:2: Löschen Sie vorhandene Serien, definieren Sie drei Kategorienamen und fügen Sie dem Diagramm eine Serie mit Beispieldaten hinzu.
//ExStepSummary:3: Setzen Sie die Strichstärke des Plot‑Bereichs auf Null, wodurch die Begrenzung entfernt wird, während die Achsenlinien sichtbar bleiben.
//ExStepSummary:4: Speichern Sie das geänderte Dokument als .docx‑Datei.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart columnChart = chartContainer.Chart;

//ExStep:2-
columnChart.Series.Clear();
string[] labels = { "Group A", "Group B", "Group C" };
columnChart.Series.Add("Demo Data", labels, new double[] { 10, 20, 30 });

//ExStep:3-
columnChart.Format.Stroke.Weight = 0;

//ExStep:4-
wordDoc.Save("ChartWithoutPlotBorder.docx");
//ExEnd
