//ExSummary: 本教程演示如何创建一个新的 Word 文档，配置具有特定字体设置的自定义文本水印，将水印应用到文档，并使用 Document.Watermark.SetText API 将结果保存为 .docx 文件。
//ExStepSummary:0: 创建一个名为 `document` 的新的空 Word 文档实例。
//ExStepSummary:1: 使用 Arial 48 磅灰色字体、对角布局和不透明渲染来配置 `watermarkSettings`。
//ExStepSummary:2: 使用先前定义的设置，将文本水印 "Private" 应用于 `document`。
//ExStepSummary:3: 定义保存带水印文档的文件路径。
//ExStepSummary:4: 将修改后的 `document` 保存到指定路径为 .docx 文件。
//ExStepImage:2:images/1.png
//ExStep:0-
Document document = new Document();

//ExStep:1-
TextWatermarkOptions watermarkSettings = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Gray,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:2-
document.Watermark.SetText("Private", watermarkSettings);

//ExStep:3-
string outputPath = "SampleWatermarked.docx";

//ExStep:4-
document.Save(outputPath);
//ExEnd
