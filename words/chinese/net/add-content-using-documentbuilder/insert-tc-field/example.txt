//ExSummary: 本教程演示了如何创建一个新的 Word 文档，使用 DocumentBuilder 插入 TC（目录）字段，并保存生成的文件。
//ExStepSummary:0: 定义生成的文档将要保存的文件夹路径。
//ExStepSummary:1: 实例化一个空的 Document 和一个用于编辑它的 DocumentBuilder。
//ExStepSummary:2: 插入一个 TC 字段，条目文本为 "Entry Text"，该文本将用于目录。
//ExStepSummary:3: 将修改后的文档保存到指定的目录。
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
//ExEnd
