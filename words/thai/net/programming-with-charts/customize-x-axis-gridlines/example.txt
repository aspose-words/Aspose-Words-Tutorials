//ExStepSummary:0: สร้างเอกสาร Word ใหม่และ DocumentBuilder เพื่อแก้ไขเอกสารนั้น.
//ExStepSummary:1: แทรกแผนภูมิคอลัมน์ขนาด 432×252 ลงในเอกสารและรับอ็อบเจ็กต์ Chart ของมัน.
//ExStepSummary:2: ลบซีรีส์เริ่มต้นและเพิ่มซีรีส์ที่กำหนดเองที่มีสามกลุ่มและค่าต่าง ๆ.
//ExStepSummary:3: รับอ้างอิงถึงแกนแนวนอน (X) หลักของแผนภูมิ.
//ExStepSummary:4: เปิดการแสดงเส้นกริดหลักบนแกน X หลัก.
//ExStepSummary:5: ตั้งค่าสีของเส้นกริดเป็นสีฟ้าและความหนาเป็น 2 จุด.
//ExStepSummary:6: บันทึกเอกสารเป็นไฟล์ .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
