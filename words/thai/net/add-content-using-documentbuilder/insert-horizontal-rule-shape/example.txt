//ExStepSummary:0: สร้างเอกสาร Word ว่างใหม่โดยการสร้างอินสแตนซ์ของคลาส Document.
//ExStepSummary:1: สร้าง DocumentBuilder ที่เชื่อมต่อกับเอกสารเพื่อเปิดใช้งานการดำเนินการแก้ไข.
//ExStepSummary:2: เขียนย่อหน้าข้อความแล้วแทรกรูปร่างเส้นกั้นแนวนอนที่ตำแหน่งเคอร์เซอร์โดยใช้ DocumentBuilder.
//ExStepSummary:3: กำหนดเส้นทางโฟลเดอร์ผลลัพธ์และบันทึกเอกสารเป็นไฟล์ .docx.
//ExStepImage:2:images/1.png
//ExStep:0-
Document doc = new Document();

//ExStep:1-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.Writeln("Insert a horizontal rule shape into the document.");
builder.InsertHorizontalRule();

//ExStep:3-
string dataDir = "YOUR DOCUMENT DIRECTORY";
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHorizontalRule.docx");
