//ExStepSummary:0: กำหนดเส้นทางโฟลเดอร์ที่เอกสารที่สร้างขึ้นจะถูกบันทึก
//ExStepSummary:1: สร้างอินสแตนซ์ของ Document ว่างเปล่าและ DocumentBuilder ที่จะทำการแก้ไขมัน
//ExStepSummary:2: แทรกฟิลด์ TC พร้อมข้อความรายการ "Entry Text" ที่จะใช้สำหรับสารบัญ
//ExStepSummary:3: บันทึกเอกสารที่แก้ไขแล้วไปยังไดเรกทอรีที่ระบุ
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
