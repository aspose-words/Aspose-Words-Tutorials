//ExSummary: Ten tutorial pokazuje, jak automatycznie dodać czerwoną przekątną znak wodny tekstowy do każdego nowego dokumentu Word generowanego w procesie raportowania wsadowego przy użyciu Aspose.Words Document i DocumentBuilder.
//ExStepSummary:0: Utwórz folder "GeneratedReports", w którym będą zapisywane pliki wyjściowe.
//ExStepSummary:1: Rozpocznij pętlę, która wygeneruje trzy oddzielne dokumenty.
//ExStepSummary:2: Utwórz nowy pusty obiekt dokumentu Word.
//ExStepSummary:3: Użyj DocumentBuilder, aby zapisać w dokumencie linię tytułu i opis.
//ExStepSummary:4: Zdefiniuj wygląd znaku wodnego, w tym czcionkę, rozmiar, kolor i układ przekątny.
//ExStepSummary:5: Zastosuj skonfigurowany czerwony przekątny znak wodny z tekstem "PROTECTED" w dokumencie.
//ExStepSummary:6: Zapisz oznaczony znak wodny dokument w folderze "GeneratedReports" pod unikalną nazwą pliku.
//ExStepSummary:7: Zamknij pętlę po przetworzeniu bieżącego dokumentu.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
