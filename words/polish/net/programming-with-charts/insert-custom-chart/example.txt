//ExSummary: Ten samouczek pokazuje, jak utworzyć dokument Word, wstawić wykres słupkowy, wyczyścić jego domyślne dane i wypełnić je własnymi danymi z dwuwymiarowej tablicy — mapując każdy wiersz na serię i każdą kolumnę na kategorię — a następnie dodać tytuł i zapisać plik.
//ExStepSummary:0: Utwórz nowy Document oraz DocumentBuilder podłączony do niego.
//ExStepSummary:1: Wstaw wykres słupkowy o rozmiarze 432 × 252 punktów do dokumentu i uzyskaj jego obiekt Chart.
//ExStepSummary:2: Usuń automatycznie wygenerowaną przykładową serię z wykresu.
//ExStepSummary:3: Zdefiniuj etykiety kategorii osi X oraz pozycje legendy dla każdej serii.
//ExStepSummary:4: Utwórz dwuwymiarową tablicę, w której każdy wiersz zawiera wartości jednej serii.
//ExStepSummary:5: Iteruj po każdej serii, wyodrębnij jej wartości wiersza z macierzy i dodaj serię do wykresu wraz z odpowiednią pozycją legendy oraz etykietami osi X.
//ExStepSummary:6: Ustaw tekst tytułu wykresu na „Half‑Yearly Revenue” i spraw, aby był widoczny.
//ExStepSummary:7: Zapisz dokument zawierający własny wykres do pliku .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:5:images/3.png
//ExStepImage:6:images/4.png
//ExStep:0-
Document myDocument = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDocument);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chartObj = chartContainer.Chart;

//ExStep:2-
chartObj.Series.Clear();

//ExStep:3-
string[] xLabels = { "Jan", "Feb", "Mar", "Apr" };
string[] legendEntries = { "Item Alpha", "Item Beta", "Item Gamma" };

//ExStep:4-
double[,] dataMatrix = {
    { 120.5, 135.0, 150.2, 165.3 }, // Item Alpha
    { 80.0,  95.5, 110.1, 130.0 }, // Item Beta
    { 60.3,  70.4,  85.6,  95.2 }   // Item Gamma
};

//ExStep:5-
for (int sIdx = 0; sIdx < legendEntries.Length; sIdx++)
{
    double[] rowValues = new double[xLabels.Length];
    for (int cIdx = 0; cIdx < xLabels.Length; cIdx++)
    {
        rowValues[cIdx] = dataMatrix[sIdx, cIdx];
    }

    chartObj.Series.Add(legendEntries[sIdx], xLabels, rowValues);
}

//ExStep:6-
chartObj.Title.Text = "Half‑Yearly Revenue";
chartObj.Title.Show = true;

//ExStep:7-
myDocument.Save("CustomChart.docx");
//ExEnd
