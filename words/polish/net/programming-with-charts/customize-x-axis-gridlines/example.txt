//ExStepSummary:0: Utwórz nowy dokument Word i obiekt DocumentBuilder, aby go edytować.
//ExStepSummary:1: Wstaw wykres słupkowy o wymiarach 432×252 do dokumentu i uzyskaj jego obiekt Chart.
//ExStepSummary:2: Usuń domyślną serię i dodaj własną serię z trzema grupami i wartościami.
//ExStepSummary:3: Uzyskaj odwołanie do pierwotnej poziomej osi (X) wykresu.
//ExStepSummary:4: Włącz wyświetlanie głównych linii siatki na pierwotnej osi X.
//ExStepSummary:5: Ustaw kolor linii siatki na niebieski oraz ich grubość na 2 punkty.
//ExStepSummary:6: Zapisz dokument jako plik .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
