//ExStepSummary:2: Zainicjalizuj nowy obiekt Document oraz DocumentBuilder powiązany z nim.
//ExStepSummary:3: Wywołaj `builder.InsertComboBox`, aby umieścić combo box o nazwie "DropDown" z określonymi elementami w bieżącej pozycji kursora.

//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
string[] items = { "One", "Two", "Three" };

//ExStep:2-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:3-
builder.InsertComboBox("DropDown", items, 0);

//ExStep:4-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertComboBoxFormField.docx");
