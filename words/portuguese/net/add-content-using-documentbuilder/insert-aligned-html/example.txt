//ExStart
//ExSummary: Usando Aspose.Words Document e DocumentBuilder, este tutorial demonstra como inserir conteúdo HTML bruto em um documento Word.
//ExStepSummary:0: Defina o caminho da pasta onde o documento resultante será salvo.
//ExStepSummary:1: Crie uma nova instância de documento Word vazia.
//ExStepSummary:2: Instancie um DocumentBuilder vinculado ao documento recém‑criado.
//ExStepSummary:3: Chame `builder.InsertHtml()` para incorporar a marcação HTML fornecida como conteúdo formatado do Word.
//ExStepSummary:4: Salve o documento modificado no diretório especificado.
//ExStepImage:3:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();

//ExStep:2-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:3-
builder.InsertHtml(
    "<P align='right'>Paragraph right</P>" +
    "<b>Implicit paragraph left</b>" +
    "<div align='center'>Div center</div>" +
    "<h1 align='left'>Heading 1 left.</h1>");

//ExStep:4-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHtml.docx");
//ExEnd
