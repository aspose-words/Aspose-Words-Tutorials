//ExSummary: Este tutorial mostra como criar um documento Word, inserir um gráfico de colunas, substituir seus dados de exemplo e, em seguida, habilitar e estilizar as linhas de grade principais no eixo X primário.
//ExStepSummary:0: Crie um novo documento Word e um DocumentBuilder para editá‑lo.
//ExStepSummary:1: Insira um gráfico de colunas 432×252 no documento e obtenha seu objeto Chart.
//ExStepSummary:2: Remova a série padrão e adicione uma série personalizada com três grupos e valores.
//ExStepSummary:3: Obtenha uma referência ao eixo horizontal (X) primário do gráfico.
//ExStepSummary:4: Habilite a exibição das linhas de grade principais no eixo X primário.
//ExStepSummary:5: Defina a cor das linhas de grade como azul e sua espessura para 2 pontos.
//ExStepSummary:6: Salve o documento em um arquivo .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
