//ExSummary: Este tutorial demonstra como criar um documento Word com um gráfico de linhas, adicionar um eixo X secundário, definir sua unidade de exibição para milhares e aplicar um formato numérico personalizado aos rótulos do eixo usando Aspose.Words.
//ExStepSummary:0: Crie um novo Document e um DocumentBuilder para começar a construir o arquivo Word.
//ExStepSummary:1: Insira um gráfico de linhas com tamanho 500×300 no documento e obtenha o objeto Chart.
//ExStepSummary:2: Remova a série de demonstração padrão que o Aspose.Words adiciona automaticamente.
//ExStepSummary:3: Defina um array de rótulos de categoria que será usado para o eixo X.
//ExStepSummary:4: Adicione a série de dados primária chamada "Main Data" usando os rótulos e valores.
//ExStepSummary:5: Crie um grupo de séries secundário do tipo Line e atribua-o ao grupo de eixo secundário.
//ExStepSummary:6: Defina a unidade de exibição do eixo X secundário para milhares, de modo que os valores sejam mostrados em K.
//ExStepSummary:7: Aplique o formato numérico personalizado "#,##0.0" aos rótulos de marcação do eixo X secundário.
//ExStepSummary:8: Opcionalmente, adicione uma série de dados auxiliar ao grupo secundário.
//ExStepSummary:9: Salve o documento em um arquivo .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStepImage:6:images/5.png
//ExStepImage:7:images/6.png
//ExStepImage:8:images/7.png
//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
