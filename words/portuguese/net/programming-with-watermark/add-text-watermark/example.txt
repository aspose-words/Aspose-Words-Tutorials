//ExSummary: Este tutorial mostra como adicionar automaticamente uma marca d'água de texto diagonal vermelha a cada novo documento Word gerado em um processo de relatório em lote usando Aspose.Words Document e DocumentBuilder.
//ExStepSummary:0: Crie a pasta "GeneratedReports" onde os arquivos de saída serão salvos.
//ExStepSummary:1: Inicie um loop que gerará três documentos separados.
//ExStepSummary:2: Crie um novo objeto de documento Word vazio.
//ExStepSummary:3: Use DocumentBuilder para escrever uma linha de título e uma descrição no documento.
//ExStepSummary:4: Defina a aparência da marca d'água, incluindo fonte, tamanho, cor e layout diagonal.
//ExStepSummary:5: Aplique a marca d'água diagonal vermelha configurada com o texto "PROTECTED" ao documento.
//ExStepSummary:6: Salve o documento com marca d'água na pasta "GeneratedReports" com um nome de arquivo exclusivo.
//ExStepSummary:7: Feche o loop após processar o documento atual.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
