//ExSummary: 使用 Aspose.Words Document 與 DocumentBuilder，本教學示範如何將原始 HTML 內容插入 Word 文件中。
//ExStepSummary:0: 定義將儲存最終文件的資料夾路徑。
//ExStepSummary:1: 建立一個新的、空白的 Word 文件實例。
//ExStepSummary:2: 實例化一個與新建立文件關聯的 DocumentBuilder。
//ExStepSummary:3: 呼叫 `builder.InsertHtml()` 以將提供的 HTML 標記嵌入為已格式化的 Word 內容。
//ExStepSummary:4: 將已修改的文件儲存至指定目錄。
//ExStepImage:3:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();

//ExStep:2-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:3-
builder.InsertHtml(
    "<P align='right'>Paragraph right</P>" +
    "<b>Implicit paragraph left</b>" +
    "<div align='center'>Div center</div>" +
    "<h1 align='left'>Heading 1 left.</h1>");

//ExStep:4-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHtml.docx");
//ExEnd
