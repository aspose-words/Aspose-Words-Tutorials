//ExSummary: Den här handledningen visar hur man skapar ett nytt Word-dokument, använder DocumentBuilder för att infoga ett TC-fält (Table of Contents) och sparar den resulterande filen.
//ExStepSummary:0: Ange sökvägen till mappen där det genererade dokumentet ska sparas.
//ExStepSummary:1: Instansiera ett tomt Document och en DocumentBuilder som kommer att redigera det.
//ExStepSummary:2: Infoga ett TC-fält med posttexten \"Entry Text\" som kommer att användas för innehållsförteckningen.
//ExStepSummary:3: Spara det modifierade dokumentet till den angivna katalogen.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
//ExEnd
