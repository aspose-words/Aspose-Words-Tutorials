//ExSummary: Den här handledningen visar hur man skapar ett Word‑dokument, infogar ett stapeldiagram, ersätter dess exempeldata och sedan aktiverar och formaterar huvudrutnätslinjerna på den primära X‑axeln.
//ExStepSummary:0: Skapa ett nytt Word‑dokument och en DocumentBuilder för att redigera det.
//ExStepSummary:1: Infoga ett 432×252 stapeldiagram i dokumentet och hämta dess Chart‑objekt.
//ExStepSummary:2: Ta bort standardserien och lägg till en anpassad serie med tre grupper och värden.
//ExStepSummary:3: Hämta en referens till diagrammets primära horisontella (X‑)axel.
//ExStepSummary:4: Aktivera visning av huvudrutnätslinjer på den primära X‑axeln.
//ExStepSummary:5: Ställ in rutnätslinjernas färg till blå och deras tjocklek till 2 punkter.
//ExStepSummary:6: Spara dokumentet som en .docx‑fil.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
