//ExStepSummary:4: Skapa en sekundär series group av typen Line och tilldela den till den sekundära axelgruppen.

//ExStepSummary:6: Visa titeln för den sekundära Y‑axeln och sätt dess text till "Sales (USD)".
//ExStepSummary:7: Ställ in nummerformatet för den sekundära Y‑axeln till ett valutamönster med två decimaler och koppla bort det från källformatet.
//ExStepSummary:8: Lägg till en hjälpserie i den sekundära gruppen som tillhandahåller data som ska plottas mot den sekundära Y‑axeln.

//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape lineChartShape = docBuilder.InsertChart(ChartType.Line, 450, 250);
Chart lineChart = lineChartShape.Chart;

//ExStep:2-
lineChart.Series.Clear();

//ExStep:3-
string[] quarters = new[] { "Q1", "Q2", "Q3", "Q4" };
lineChart.Series.Add("Main Series", quarters, new double[] { 1200, 1500, 1800, 2100 });

//ExStep:4-
ChartSeriesGroup secondarySeriesGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
secondarySeriesGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:5-
secondarySeriesGroup.AxisX.Hidden = true;

//ExStep:6-
secondarySeriesGroup.AxisY.Title.Show = true;
secondarySeriesGroup.AxisY.Title.Text = "Sales (USD)";

//ExStep:7-
secondarySeriesGroup.AxisY.NumberFormat.FormatCode = "\"$\"#,##0.00";
secondarySeriesGroup.AxisY.NumberFormat.IsLinkedToSource = false;

//ExStep:8-
secondarySeriesGroup.Series.Add("Auxiliary Series", quarters, new double[] { 3000, 3500, 4000, 4500 });

//ExStep:9-
string resultPath = Path.Combine(Directory.GetCurrentDirectory(), "ProfitChart.docx");
wordDoc.Save(resultPath);
