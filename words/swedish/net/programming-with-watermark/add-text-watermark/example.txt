//ExSummary: Den här handledningen visar hur du automatiskt lägger till ett rött diagonalt textvattenmärke i varje nytt Word-dokument som genereras i en batchrapportprocess med hjälp av Aspose.Words Document och DocumentBuilder.
//ExStepSummary:0: Skapa mappen "GeneratedReports" där utdatafilerna kommer att sparas.
//ExStepSummary:1: Starta en loop som kommer att generera tre separata dokument.
//ExStepSummary:2: Skapa ett nytt tomt Word-dokumentobjekt.
//ExStepSummary:3: Använd DocumentBuilder för att skriva en titelrad och en beskrivning i dokumentet.
//ExStepSummary:4: Definiera utseendet på vattenmärket, inklusive teckensnitt, storlek, färg och diagonal layout.
//ExStepSummary:5: Applicera det konfigurerade röda diagonala vattenmärket med texten "PROTECTED" på dokumentet.
//ExStepSummary:6: Spara det vattenmärkta dokumentet i mappen "GeneratedReports" med ett unikt filnamn.
//ExStepSummary:7: Stäng loopen efter att ha bearbetat det aktuella dokumentet.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
