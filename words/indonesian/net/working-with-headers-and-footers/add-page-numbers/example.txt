//ExStart
//ExSummary: Menggunakan Aspose.Words Document dan DocumentBuilder, tutorial ini menambahkan nomor halaman yang otomatis diperbarui ke footer utama dokumen Word.
//ExStepSummary:0: Buat objek Document baru dan DocumentBuilder yang terhubung dengannya.
//ExStepSummary:1: Pindahkan kursor builder ke footer utama pada bagian pertama.
//ExStepSummary:2: Atur perataan paragraf ke tengah sehingga teks footer akan berada di tengah.
//ExStepSummary:3: Tuliskan label \"Page \" dan sisipkan field PAGE yang menampilkan nomor halaman saat ini.
//ExStepSummary:4: Tuliskan \" of \" dan sisipkan field NUMPAGES yang menunjukkan total jumlah halaman.
//ExStepSummary:5: Simpan dokumen ke file .docx.
//ExStepImage:3:images/1.png
//ExStepImage:4:images/2.png
//ExStep:0-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:1-
builder.MoveToHeaderFooter(HeaderFooterType.FooterPrimary);

//ExStep:2-
builder.ParagraphFormat.Alignment = ParagraphAlignment.Center;

//ExStep:3-
builder.Write("Page ");
builder.InsertField("PAGE", null);

//ExStep:4-
builder.Write(" of ");
builder.InsertField("NUMPAGES", null);

//ExStep:5-
doc.Save("PageNumbers.docx");
//ExEnd
