//ExSummary: Tutorial ini menunjukkan cara membuat dokumen Word, menyisipkan diagram kolom, mengganti data contoh, dan kemudian mengaktifkan serta menata garis kisi utama pada sumbu X utama.
//ExStepSummary:0: Buat dokumen Word baru dan DocumentBuilder untuk mengeditnya.
//ExStepSummary:1: Sisipkan diagram kolom berukuran 432×252 ke dalam dokumen dan dapatkan objek Chart-nya.
//ExStepSummary:2: Hapus seri default dan tambahkan seri khusus dengan tiga grup dan nilai.
//ExStepSummary:3: Dapatkan referensi ke sumbu horizontal (X) utama chart.
//ExStepSummary:4: Aktifkan tampilan garis kisi utama pada sumbu X utama.
//ExStepSummary:5: Atur warna garis kisi menjadi biru dan ketebalannya menjadi 2 poin.
//ExStepSummary:6: Simpan dokumen ke file .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document myDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();
chart.Series.Add("Demo Series",
    new[] { "Group One", "Group Two", "Group Three" },
    new double[] { 10, 20, 30 });

//ExStep:3-
ChartAxis horizontalAxis = chart.AxisX;

//ExStep:4-
horizontalAxis.HasMajorGridlines = true;

//ExStep:5-
horizontalAxis.Format.Stroke.Color = Color.Blue;
horizontalAxis.Format.Stroke.Weight = 2.0;

//ExStep:6-
myDoc.Save("PrimaryXGridlines.docx");
//ExEnd
