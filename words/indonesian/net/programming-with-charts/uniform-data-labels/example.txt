//ExSummary: Tutorial ini menunjukkan cara membuat dokumen Word, menyisipkan diagram kolom, menambahkan seri data khusus, dan mendefinisikan ukuran font dan warna label data default yang diterapkan secara seragam pada semua seri dalam diagram.
//ExStepSummary:0: Buat dokumen Word kosong baru dan DocumentBuilder untuk mengeditnya.
//ExStepSummary:1: Sisipkan diagram kolom berukuran 500×300 ke dalam dokumen dan dapatkan objek Chart-nya.
//ExStepSummary:2: Hapus seri contoh default yang ditambahkan Aspose.Words ke diagram baru.
//ExStepSummary:3: Tetapkan label kategori dan tambahkan dua seri data (Item Alpha dan Item Beta) beserta nilainya ke diagram.
//ExStepSummary:4: Tetapkan ukuran font dan warna dasar yang akan diterapkan pada semua label data.
//ExStepSummary:5: Aktifkan label data untuk setiap seri, tampilkan nilai numerik, dan terapkan ukuran font serta warna yang telah ditentukan.
//ExStepSummary:6: Simpan dokumen ke file .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStepImage:5:images/4.png
//ExStep:0-
Document document = new Document();
DocumentBuilder builder = new DocumentBuilder(document);

//ExStep:1-
Shape chartContainer = builder.InsertChart(ChartType.Column, 500, 300);
Chart chart = chartContainer.Chart;

//ExStep:2-
chart.Series.Clear();

//ExStep:3-
string[] periods = { "Period 1", "Period 2", "Period 3", "Period 4" };
chart.Series.Add("Item Alpha", periods, new double[] { 120, 150, 170, 130 });
chart.Series.Add("Item Beta", periods, new double[] { 80, 110, 140, 100 });

//ExStep:4-
const float labelFontSize = 10f;
Color labelFontColor = Color.White;

//ExStep:5-
foreach (ChartSeries series in chart.Series)
{
    series.HasDataLabels = true;
    series.DataLabels.ShowValue = true;
    series.DataLabels.Font.Size = labelFontSize;
    series.DataLabels.Font.Color = labelFontColor;
}

//ExStep:6-
document.Save("UniformLabelChart.docx");
//ExEnd
