//ExSummary: Tutorial ini menunjukkan cara membuat dokumen Word, menyisipkan diagram kolom, mengisinya dengan data contoh, menyembunyikan batas area plot diagram sambil mempertahankan garis sumbu tetap terlihat, dan akhirnya menyimpan dokumen.
//ExStepSummary:0: Buat dokumen Word kosong baru dan DocumentBuilder untuk mengeditnya.
//ExStepSummary:1: Sisipkan diagram kolom berukuran 432×252 ke dalam dokumen dan dapatkan objek Chart-nya.
//ExStepSummary:2: Bersihkan semua seri yang ada, tentukan tiga label kategori, dan tambahkan satu seri dengan nilai contoh ke diagram.
//ExStepSummary:3: Atur ketebalan stroke area plot menjadi nol, menghapus batasnya sambil membiarkan garis sumbu tetap terlihat.
//ExStepSummary:4: Simpan dokumen yang telah dimodifikasi ke file .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart columnChart = chartContainer.Chart;

//ExStep:2-
columnChart.Series.Clear();
string[] labels = { "Group A", "Group B", "Group C" };
columnChart.Series.Add("Demo Data", labels, new double[] { 10, 20, 30 });

//ExStep:3-
columnChart.Format.Stroke.Weight = 0;

//ExStep:4-
wordDoc.Save("ChartWithoutPlotBorder.docx");
//ExEnd
