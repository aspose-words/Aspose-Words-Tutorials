//ExStepSummary:0: Instansiasi Document baru dan DocumentBuilder yang akan digunakan untuk membangun tabel.

//ExStepSummary:2: Ratakan konten secara vertikal di tengah pada sel saat ini dan tulis teks sel pertama baris pertama.
//ExStepSummary:3: Sisipkan sel kedua pada baris pertama dan tulis teksnya.
//ExStepSummary:4: Tutup baris pertama, menyelesaikan tata letaknya.
//ExStepSummary:5: Mulai sel pertama pada baris kedua, atur tinggi baris menjadi tepat 100 poin, putar teks ke atas, dan tulis teks sel tersebut.
//ExStepSummary:6: Sisipkan sel kedua pada baris kedua, putar teksnya ke bawah, dan tulis teks sel tersebut.
//ExStepSummary:7: Tutup baris kedua, menyelesaikan baris kedua tabel.
//ExStepSummary:8: Akiri konstruksi tabel, mengunci struktur tabel.

//ExStep:0-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:1-
Table table = builder.StartTable();
builder.InsertCell();
table.AutoFit(AutoFitBehavior.FixedColumnWidths);

//ExStep:2-
builder.CellFormat.VerticalAlignment = CellVerticalAlignment.Center;
builder.Write("This is row 1 cell 1");

//ExStep:3-
builder.InsertCell();
builder.Write("This is row 1 cell 2");

//ExStep:4-
builder.EndRow();

//ExStep:5-
builder.InsertCell();

builder.RowFormat.Height = 100;
builder.RowFormat.HeightRule = HeightRule.Exactly;
builder.CellFormat.Orientation = TextOrientation.Upward;
builder.Writeln("This is row 2 cell 1");

//ExStep:6-
builder.InsertCell();
builder.CellFormat.Orientation = TextOrientation.Downward;
builder.Writeln("This is row 2 cell 2");

//ExStep:7-
builder.EndRow();

//ExStep:8-
builder.EndTable();

//ExStep:9-
doc.Save("YourDirectoryPath/AddContentUsingDocumentBuilder.BuildTable.docx");
