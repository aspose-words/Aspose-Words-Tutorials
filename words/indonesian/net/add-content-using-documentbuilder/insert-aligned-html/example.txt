//ExSummary: Dengan menggunakan Aspose.Words Document dan DocumentBuilder, tutorial ini menunjukkan cara menyisipkan konten HTML mentah ke dalam dokumen Word.
//ExStepSummary:0: Definisikan jalur folder tempat dokumen hasil akan disimpan.
//ExStepSummary:1: Buat instance dokumen Word baru yang kosong.
//ExStepSummary:2: Buat instance DocumentBuilder yang terhubung ke dokumen yang baru dibuat.
//ExStepSummary:3: Panggil `builder.InsertHtml()` untuk menyisipkan markup HTML yang diberikan sebagai konten Word yang diformat.
//ExStepSummary:4: Simpan dokumen yang telah dimodifikasi ke direktori yang ditentukan.
//ExStepImage:3:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();

//ExStep:2-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:3-
builder.InsertHtml(
    "<P align='right'>Paragraph right</P>" +
    "<b>Implicit paragraph left</b>" +
    "<div align='center'>Div center</div>" +
    "<h1 align='left'>Heading 1 left.</h1>");

//ExStep:4-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHtml.docx");
//ExEnd
