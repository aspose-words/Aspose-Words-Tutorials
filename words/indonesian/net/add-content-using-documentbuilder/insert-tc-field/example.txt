//ExSummary: Tutorial ini menunjukkan cara membuat dokumen Word baru, menggunakan DocumentBuilder untuk menyisipkan field TC (Table of Contents), dan menyimpan file yang dihasilkan.
//ExStepSummary:0: Tentukan jalur folder tempat dokumen yang dihasilkan akan disimpan.
//ExStepSummary:1: Buat instance Document kosong dan DocumentBuilder yang akan mengeditnya.
//ExStepSummary:2: Sisipkan field TC dengan teks entri \"Entry Text\" yang akan digunakan untuk daftar isi.
//ExStepSummary:3: Simpan dokumen yang telah dimodifikasi ke direktori yang ditentukan.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
//ExEnd
