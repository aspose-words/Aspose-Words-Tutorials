//ExSummary: Tutorial ini menunjukkan cara membuat dokumen Word, menambahkan paragraf teks, dan menyisipkan bentuk garis horizontal menggunakan Document dan DocumentBuilder.
//ExStepSummary:0: Buat dokumen Word kosong baru dengan menginstansiasi kelas Document.
//ExStepSummary:1: Buat DocumentBuilder yang terhubung ke dokumen untuk memungkinkan operasi penyuntingan.
//ExStepSummary:2: Tulis paragraf teks lalu sisipkan bentuk garis horizontal pada posisi kursor menggunakan DocumentBuilder.
//ExStepSummary:3: Tentukan jalur folder output dan simpan dokumen ke file .docx.
//ExStepImage:2:images/1.png
//ExStep:0-
Document doc = new Document();

//ExStep:1-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.Writeln("Insert a horizontal rule shape into the document.");
builder.InsertHorizontalRule();

//ExStep:3-
string dataDir = "YOUR DOCUMENT DIRECTORY";
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHorizontalRule.docx");
