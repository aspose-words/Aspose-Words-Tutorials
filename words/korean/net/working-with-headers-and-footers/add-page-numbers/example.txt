//ExSummary: Aspose.Words Document와 DocumentBuilder를 사용하여, 이 튜토리얼은 Word 문서의 기본 푸터에 자동으로 업데이트되는 페이지 번호를 추가합니다.
//ExStepSummary:0: 새 Document 객체와 이에 연결된 DocumentBuilder를 생성합니다.
//ExStepSummary:1: builder의 커서를 첫 번째 섹션의 기본 푸터로 이동합니다.
//ExStepSummary:2: 단락 정렬을 가운데로 설정하여 푸터 텍스트가 중앙에 배치되도록 합니다.
//ExStepSummary:3: "Page " 라벨을 쓰고 현재 페이지 번호를 표시하는 PAGE 필드를 삽입합니다.
//ExStepSummary:4: " of " 를 쓰고 전체 페이지 수를 표시하는 NUMPAGES 필드를 삽입합니다.
//ExStepSummary:5: 문서를 .docx 파일로 저장합니다.
//ExStepImage:3:images/1.png
//ExStepImage:4:images/2.png
//ExStep:0-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:1-
builder.MoveToHeaderFooter(HeaderFooterType.FooterPrimary);

//ExStep:2-
builder.ParagraphFormat.Alignment = ParagraphAlignment.Center;

//ExStep:3-
builder.Write("Page ");
builder.InsertField("PAGE", null);

//ExStep:4-
builder.Write(" of ");
builder.InsertField("NUMPAGES", null);

//ExStep:5-
doc.Save("PageNumbers.docx");
//ExEnd
