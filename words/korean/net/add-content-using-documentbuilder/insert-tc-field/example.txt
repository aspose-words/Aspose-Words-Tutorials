//ExSummary: 이 튜토리얼에서는 새 Word 문서를 만들고, DocumentBuilder를 사용하여 TC(목차) 필드를 삽입한 다음, 결과 파일을 저장하는 방법을 보여줍니다.
//ExStepSummary:0: 생성된 문서를 저장할 폴더 경로를 정의합니다.
//ExStepSummary:1: 빈 Document와 이를 편집할 DocumentBuilder를 인스턴스화합니다.
//ExStepSummary:2: 목차에 사용할 항목 텍스트 "Entry Text"가 포함된 TC 필드를 삽입합니다.
//ExStepSummary:3: 수정된 문서를 지정된 디렉터리에 저장합니다.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
//ExEnd
