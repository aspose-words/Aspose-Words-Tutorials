//ExSummary: 이 튜토리얼에서는 Document와 DocumentBuilder를 사용하여 Word 문서를 만들고, 텍스트 단락을 추가한 뒤, 가로 구분선 형태를 삽입하는 방법을 보여줍니다.
//ExStepSummary:0: Document 클래스를 인스턴스화하여 새 빈 Word 문서를 생성합니다.
//ExStepSummary:1: 문서와 연결된 DocumentBuilder를 생성하여 편집 작업을 수행할 수 있게 합니다.
//ExStepSummary:2: 텍스트 단락을 작성한 다음 DocumentBuilder를 사용하여 커서 위치에 가로 구분선 형태를 삽입합니다.
//ExStepSummary:3: 출력 폴더 경로를 정의하고 문서를 .docx 파일로 저장합니다.
//ExStepImage:2:images/1.png
//ExStep:0-
Document doc = new Document();

//ExStep:1-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.Writeln("Insert a horizontal rule shape into the document.");
builder.InsertHorizontalRule();

//ExStep:3-
string dataDir = "YOUR DOCUMENT DIRECTORY";
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHorizontalRule.docx");
