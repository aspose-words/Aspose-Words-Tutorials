//ExSummary: 이 튜토리얼에서는 Word 문서를 생성하고, 열 차트를 삽입한 뒤 기본 데이터를 지우고, 2차원 배열의 사용자 정의 데이터로 채워 각 행을 시리즈, 각 열을 카테고리로 매핑하는 방법을 보여줍니다. 그런 다음 제목을 추가하고 파일을 저장합니다.
//ExStepSummary:0: 새 Document를 만들고, 해당 Document에 연결된 DocumentBuilder를 생성합니다.
//ExStepSummary:1: 문서에 크기 432x252 포인트인 열 차트를 삽입하고 해당 Chart 객체를 가져옵니다.
//ExStepSummary:2: 차트에서 자동으로 생성된 샘플 시리즈를 제거합니다.
//ExStepSummary:3: X축 카테고리 레이블과 각 시리즈에 대한 범례 항목을 정의합니다.
//ExStepSummary:4: 각 행이 하나의 시리즈 값을 보관하는 2차원 배열을 생성합니다.
//ExStepSummary:5: 각 시리즈를 순회하면서 매트릭스에서 해당 행 값을 추출하고, 해당 범례 항목과 X축 레이블을 사용하여 차트에 시리즈를 추가합니다.
//ExStepSummary:6: 차트 제목 텍스트를 'Half‑Yearly Revenue'로 설정하고 표시하도록 합니다.
//ExStepSummary:7: 사용자 정의 차트가 포함된 문서를 .docx 파일로 저장합니다.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:5:images/3.png
//ExStepImage:6:images/4.png
//ExStep:0-
Document myDocument = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(myDocument);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart chartObj = chartContainer.Chart;

//ExStep:2-
chartObj.Series.Clear();

//ExStep:3-
string[] xLabels = { "Jan", "Feb", "Mar", "Apr" };
string[] legendEntries = { "Item Alpha", "Item Beta", "Item Gamma" };

//ExStep:4-
double[,] dataMatrix = {
    { 120.5, 135.0, 150.2, 165.3 }, // Item Alpha
    { 80.0,  95.5, 110.1, 130.0 }, // Item Beta
    { 60.3,  70.4,  85.6,  95.2 }   // Item Gamma
};

//ExStep:5-
for (int sIdx = 0; sIdx < legendEntries.Length; sIdx++)
{
    double[] rowValues = new double[xLabels.Length];
    for (int cIdx = 0; cIdx < xLabels.Length; cIdx++)
    {
        rowValues[cIdx] = dataMatrix[sIdx, cIdx];
    }

    chartObj.Series.Add(legendEntries[sIdx], xLabels, rowValues);
}

//ExStep:6-
chartObj.Title.Text = "Half‑Yearly Revenue";
chartObj.Title.Show = true;

//ExStep:7-
myDocument.Save("CustomChart.docx");
//ExEnd
