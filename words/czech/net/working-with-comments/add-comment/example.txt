//ExSummary: Tento tutoriál ukazuje, jak vytvořit nový dokument Word, přidat tři odstavce, najít konkrétní odstavec, připojit k němu komentář s informacemi o autorovi a výsledek uložit jako soubor DOCX pomocí Aspose.Words.
//ExStepSummary:0: Vytvořte novou instanci třídy Document a DocumentBuilder, abyste mohli začít sestavovat soubor Word.
//ExStepSummary:1: Použijte DocumentBuilder.Writeln k přidání tří samostatných odstavců textu do dokumentu.
//ExStepSummary:2: Bezpečně načtěte druhý odstavec, přičemž v případě, že dokument obsahuje méně než dva odstavce, použijte první.
//ExStepSummary:3: Vytvořte Comment s údaji o autorovi, nastavte jeho text a připojte jej k vybranému odstavci.
//ExStepSummary:4: Uložte upravený dokument jako soubor DOCX s názvem 'sample.docx' do aktuální složky.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
