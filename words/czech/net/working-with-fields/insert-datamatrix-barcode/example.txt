//ExSummary: Tento tutoriál ukazuje, jak vytvořit nový dokument Word, vložit pole DISPLAYBARCODE, nastavit jej pro čárový kód DataMatrix a vykreslit obrázek čárového kódu pomocí tříd Document a DocumentBuilder.
//ExStepSummary:0: Vytvořte nový prázdný dokument Word a DocumentBuilder pro jeho úpravu.
//ExStepSummary:1: Vložte pole DISPLAYBARCODE na aktuální pozici kurzoru, čímž se do dokumentu přidá zástupce pole.
//ExStepSummary:2: Nastavte vlastnost BarcodeType pole na DataMatrix a zadejte řetězec dat k zakódování.
//ExStepSummary:3: Volitelně definujte barvy pozadí a popředí čárového kódu.
//ExStepSummary:4: Zavolejte na dokumentu metodu UpdateFields, aby se v poli vykreslil obrázek čárového kódu.
//ExStepSummary:5: Uložte dokument do souboru .docx.
//ExStepImage:4:images/1.png
//ExStep:0-
// Initialise an empty Word document.
Document document = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
// Add a DISPLAYBARCODE field to the document.
FieldDisplayBarcode displayBarcodeField = (FieldDisplayBarcode)docBuilder.InsertField(FieldType.FieldDisplayBarcode, true);

//ExStep:2-
// Define the barcode format as DataMatrix and assign a sample payload.
displayBarcodeField.BarcodeType = "DataMatrix";
displayBarcodeField.BarcodeValue = "9876543210";

//ExStep:3-
// (Optional) Specify background and foreground colours.
displayBarcodeField.BackgroundColor = "0xFFFFFF";
displayBarcodeField.ForegroundColor = "0x000000";

//ExStep:4-
// Refresh all fields so the barcode image is rendered.
document.UpdateFields();

//ExStep:5-
// Persist the document to a file.
document.Save("DataMatrix_BarcodeDemo.docx");
//ExEnd
