//ExStepSummary:0: Vytvořte nový Document a DocumentBuilder pro zahájení tvorby souboru Word.
//ExStepSummary:1: Vložte do dokumentu čárový graf o rozměrech 500 × 300 a získejte objekt Chart.
//ExStepSummary:2: Odstraňte výchozí demonstrační sérii, kterou Aspose.Words přidává automaticky.
//ExStepSummary:3: Definujte pole štítků kategorií, které budou použity pro osu X.
//ExStepSummary:4: Přidejte primární datovou sérii pojmenovanou „Main Data“ pomocí štítků a hodnot.
//ExStepSummary:5: Vytvořte sekundární skupinu sérií typu Line a přiřaďte ji ke skupině sekundární osy.
//ExStepSummary:6: Nastavte zobrazovací jednotku sekundární osy X na tisíce, aby se hodnoty zobrazovaly v K.
//ExStepSummary:7: Použijte vlastní číselný formát "#,##0.0" na popisky značek sekundární osy X.
//ExStepSummary:8: Volitelně přidejte pomocnou datovou sérii do sekundární skupiny.
//ExStepSummary:9: Uložte dokument do souboru .docx.

//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
