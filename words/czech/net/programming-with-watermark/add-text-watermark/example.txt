//ExSummary: Tento tutoriál ukazuje, jak automaticky přidat červený diagonální textový vodoznak do každého nového dokumentu Word generovaného v procesu hromadného reportování pomocí třídy Aspose.Words Document a DocumentBuilder.
//ExStepSummary:0: Vytvořte složku "GeneratedReports", kam budou uloženy výstupní soubory.
//ExStepSummary:1: Spusťte smyčku, která vygeneruje tři samostatné dokumenty.
//ExStepSummary:2: Vytvořte nový prázdný objekt Word dokumentu.
//ExStepSummary:3: Použijte DocumentBuilder k zápisu titulní řádky a popisu do dokumentu.
//ExStepSummary:4: Definujte vzhled vodoznaku, včetně písma, velikosti, barvy a diagonálního rozložení.
//ExStepSummary:5: Aplikujte nakonfigurovaný červený diagonální vodoznak s textem "PROTECTED" do dokumentu.
//ExStepSummary:6: Uložte dokument s vodoznakem do složky "GeneratedReports" s unikátním názvem souboru.
//ExStepSummary:7: Uzavřete smyčku po zpracování aktuálního dokumentu.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
