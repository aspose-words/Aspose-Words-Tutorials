//ExSummary: Este tutorial muestra cómo agregar automáticamente una marca de agua de texto diagonal roja a cada nuevo documento Word generado en un proceso de informe por lotes utilizando Aspose.Words Document y DocumentBuilder.
//ExStepSummary:0: Cree la carpeta "GeneratedReports" donde se guardarán los archivos de salida.
//ExStepSummary:1: Inicie un bucle que generará tres documentos separados.
//ExStepSummary:2: Cree un nuevo objeto de documento Word vacío.
//ExStepSummary:3: Utilice DocumentBuilder para escribir una línea de título y una descripción en el documento.
//ExStepSummary:4: Defina la apariencia de la marca de agua, incluyendo fuente, tamaño, color y disposición diagonal.
//ExStepSummary:5: Aplique la marca de agua diagonal roja configurada con el texto "PROTECTED" al documento.
//ExStepSummary:6: Guarde el documento con marca de agua en la carpeta "GeneratedReports" con un nombre de archivo único.
//ExStepSummary:7: Cierre el bucle después de procesar el documento actual.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
