//ExSummary: Usando Aspose.Words for .NET, este tutorial crea un Document, inserta un gráfico de columnas mediante DocumentBuilder, elimina una serie de datos específica del gráfico con ChartSeriesCollection.RemoveAt y guarda el documento modificado.
//ExStepSummary:0: Crea un nuevo Word Document y un DocumentBuilder vinculado a él.
//ExStepSummary:1: Inserta un gráfico de columnas de tamaño 432×252 puntos y obtén el objeto Chart para manipularlo posteriormente.
//ExStepSummary:2: Define el índice basado en cero de la serie que debe eliminarse (la segunda serie).
//ExStepSummary:3: Si el índice es válido, llama a RemoveAt en la colección Series del gráfico para eliminar esa serie.
//ExStepSummary:4: Guarda el documento actualizado en un archivo .docx.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart columnChart = chartContainer.Chart;

//ExStep:2-
int targetSeriesIdx = 1;

//ExStep:3-
if (targetSeriesIdx >= 0 && targetSeriesIdx < columnChart.Series.Count)
{
    columnChart.Series.RemoveAt(targetSeriesIdx);
}

//ExStep:4-
wordDoc.Save("ChartWithoutSeries.docx");
//ExEnd
