//ExSummary: Este tutorial muestra cómo crear un documento de Word con un gráfico de líneas, agregar un eje X secundario, establecer su unidad de visualización en miles y aplicar un formato numérico personalizado a las etiquetas del eje usando Aspose.Words.
//ExStepSummary:0: Cree un nuevo Document y un DocumentBuilder para comenzar a construir el archivo de Word.
//ExStepSummary:1: Inserte un gráfico de líneas de tamaño 500×300 en el documento y obtenga el objeto Chart.
//ExStepSummary:2: Elimine la serie de demostración predeterminada que Aspose.Words agrega automáticamente.
//ExStepSummary:3: Defina una matriz de etiquetas de categoría que se utilizará para el eje X.
//ExStepSummary:4: Agregue la serie de datos primaria llamada "Main Data" usando las etiquetas y los valores.
//ExStepSummary:5: Cree un grupo de series secundario de tipo Line y asígnelo al grupo de eje secundario.
//ExStepSummary:6: Establezca la unidad de visualización del eje X secundario en miles para que los valores se muestren en K.
//ExStepSummary:7: Aplique un formato numérico personalizado "#,##0.0" a las etiquetas de marcas del eje X secundario.
//ExStepSummary:8: Opcionalmente, agregue una serie de datos auxiliar al grupo secundario.
//ExStepSummary:9: Guarde el documento en un archivo .docx.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:4:images/3.png
//ExStepImage:5:images/4.png
//ExStepImage:6:images/5.png
//ExStepImage:7:images/6.png
//ExStepImage:8:images/7.png
//ExStep:0-
        Document document = new Document();
        DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
        Shape chartContainer = docBuilder.InsertChart(ChartType.Line, 500, 300);
        Chart lineChart = chartContainer.Chart;

//ExStep:2-
        lineChart.Series.Clear();

//ExStep:3-
        string[] labels = { "Group A", "Group B", "Group C" };

//ExStep:4-
        lineChart.Series.Add("Main Data", labels, new double[] { 10, 20, 30 });

//ExStep:5-
        ChartSeriesGroup secondaryGroup = lineChart.SeriesGroups.Add(ChartSeriesType.Line);
        secondaryGroup.AxisGroup = AxisGroup.Secondary;

//ExStep:6-
        secondaryGroup.AxisX.DisplayUnit.Unit = AxisBuiltInUnit.Thousands;

//ExStep:7-
        secondaryGroup.AxisX.NumberFormat.FormatCode = "#,##0.0";

//ExStep:8-
        secondaryGroup.Series.Add("Auxiliary Data", labels, new double[] { 15, 25, 35 });

//ExStep:9-
        document.Save("ChartWithSecondaryAxis.docx");
//ExEnd
