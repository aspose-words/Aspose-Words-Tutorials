//ExSummary: Este tutorial muestra cómo crear un documento Word, agregar un párrafo de texto e insertar una forma de regla horizontal usando Document y DocumentBuilder.
//ExStepSummary:0: Crea un nuevo documento Word vacío instanciando la clase Document.
//ExStepSummary:1: Crea un DocumentBuilder vinculado al documento para habilitar operaciones de edición.
//ExStepSummary:2: Escribe un párrafo de texto y luego inserta una forma de regla horizontal en la posición del cursor usando DocumentBuilder.
//ExStepSummary:3: Define la ruta de la carpeta de salida y guarda el documento en un archivo .docx.
//ExStepImage:2:images/1.png
//ExStep:0-
Document doc = new Document();

//ExStep:1-
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.Writeln("Insert a horizontal rule shape into the document.");
builder.InsertHorizontalRule();

//ExStep:3-
string dataDir = "YOUR DOCUMENT DIRECTORY";
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertHorizontalRule.docx");
//ExEnd
