//ExSummary: Questo tutorial mostra come aggiungere automaticamente una filigrana di testo rossa diagonale a ogni nuovo documento Word generato in un processo di report batch utilizzando Aspose.Words Document e DocumentBuilder.
//ExStepSummary:0: Crea la cartella "GeneratedReports" dove verranno salvati i file di output.
//ExStepSummary:1: Avvia un ciclo che genererà tre documenti separati.
//ExStepSummary:2: Crea un nuovo oggetto documento Word vuoto.
//ExStepSummary:3: Usa DocumentBuilder per scrivere una riga di titolo e una descrizione nel documento.
//ExStepSummary:4: Definisci l'aspetto della filigrana, includendo font, dimensione, colore e layout diagonale.
//ExStepSummary:5: Applica la filigrana rossa diagonale configurata con il testo "PROTECTED" al documento.
//ExStepSummary:6: Salva il documento con filigrana nella cartella "GeneratedReports" con un nome file univoco.
//ExStepSummary:7: Chiudi il ciclo dopo aver elaborato il documento corrente.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
