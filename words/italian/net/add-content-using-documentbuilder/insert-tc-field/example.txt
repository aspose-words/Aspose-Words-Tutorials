//ExSummary: Questo tutorial dimostra come creare un nuovo documento Word, utilizzare DocumentBuilder per inserire un campo TC (Table of Contents) e salvare il file risultante.
//ExStepSummary:0: Definisci il percorso della cartella in cui verrà salvato il documento generato.
//ExStepSummary:1: Istanzia un Document vuoto e un DocumentBuilder che lo modificherà.
//ExStepSummary:2: Inserisci un campo TC con il testo dell'entrata "Entry Text" che verrà utilizzato per l'indice.
//ExStepSummary:3: Salva il documento modificato nella directory specificata.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
//ExEnd
