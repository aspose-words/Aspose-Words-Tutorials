//ExSummary: Questo tutorial mostra come creare un nuovo documento Word, aggiungere tre paragrafi, individuare un paragrafo specifico, allegare un commento con le informazioni dell'autore e salvare il risultato come file DOCX utilizzando Aspose.Words.
//ExStepSummary:0: Istanziare un nuovo Document e un DocumentBuilder per iniziare a costruire il file Word.
//ExStepSummary:1: Utilizzare DocumentBuilder.Writeln per aggiungere tre paragrafi di testo separati al documento.
//ExStepSummary:2: Recuperare in modo sicuro il secondo paragrafo, ricorrendo al primo se il documento ha meno di due paragrafi.
//ExStepSummary:3: Creare un Comment con i dettagli dell'autore, impostarne il testo e allegarlo al paragrafo selezionato.
//ExStepSummary:4: Salvare il documento modificato come file DOCX denominato 'sample.docx' nella cartella corrente.
//ExStepImage:1:images/1.png
//ExStepImage:3:images/2.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder builderObj = new DocumentBuilder(wordDoc);

//ExStep:1-
builderObj.Writeln("Introductory line.");
builderObj.Writeln("Middle line.");
builderObj.Writeln("Concluding line.");

//ExStep:2-
Paragraph? chosenParagraph = null;
ParagraphCollection? paraList = wordDoc.FirstSection?.Body?.Paragraphs;
if (paraList != null && paraList.Count > 1)
{
    chosenParagraph = paraList[1];
}
else if (paraList != null && paraList.Count > 0)
{
    chosenParagraph = paraList[0];
}

//ExStep:3-
if (chosenParagraph != null)
{
    Comment note = new Comment(wordDoc, "Jordan", "JH", DateTime.Now);
    note.SetText("Please examine this section.");
    chosenParagraph.AppendChild(note);
}

//ExStep:4-
wordDoc.Save("sample.docx");
//ExEnd
