//ExSummary: Ez az útmutató bemutatja, hogyan lehet automatikusan piros átlós szöveges vízjelet hozzáadni minden új Word-dokumentumhoz, amely kötegelt jelentéskészítési folyamat során jön létre az Aspose.Words Document és DocumentBuilder használatával.
//ExStepSummary:0: Hozza létre a \"GeneratedReports\" mappát, ahová a kimeneti fájlok mentésre kerülnek.
//ExStepSummary:1: Indítson egy ciklust, amely három különálló dokumentumot generál.
//ExStepSummary:2: Hozzon létre egy új, üres Word-dokumentum objektumot.
//ExStepSummary:3: Használja a DocumentBuilder-t, hogy egy címsort és egy leírást írjon a dokumentumba.
//ExStepSummary:4: Határozza meg a vízjel megjelenését, beleértve a betűtípust, méretet, színt és az átlós elrendezést.
//ExStepSummary:5: Alkalmazza a beállított piros átlós vízjelet a \"PROTECTED\" szöveggel a dokumentumra.
//ExStepSummary:6: Mentse a vízjelezett dokumentumot a \"GeneratedReports\" mappába egy egyedi fájlnévvel.
//ExStepSummary:7: Zárja le a ciklust a jelenlegi dokumentum feldolgozása után.
//ExStepImage:3:images/1.png
//ExStepImage:5:images/2.png
//ExStep:0-
// Prepare a directory where the created files will be placed.
string reportsDir = Path.Combine(Directory.GetCurrentDirectory(), "GeneratedReports");
Directory.CreateDirectory(reportsDir);

//ExStep:1-
// Emulate the production of a few documents.
for (int index = 1; index <= 3; index++)
{

//ExStep:2-
// Instantiate a fresh, empty document.
Document document = new Document();

//ExStep:3-
// Insert some placeholder text.
DocumentBuilder docBuilder = new DocumentBuilder(document);
docBuilder.Writeln($"Document #{index}");
docBuilder.Writeln("This document was produced automatically.");

//ExStep:4-
// Configure how the watermark should look.
TextWatermarkOptions wmOptions = new TextWatermarkOptions
{
    FontFamily = "Arial",
    FontSize = 48,
    Color = Color.Red,
    Layout = WatermarkLayout.Diagonal,
    IsSemitrasparent = false
};

//ExStep:5-
// Add the text watermark to the document.
document.Watermark.SetText("PROTECTED", wmOptions);

//ExStep:6-
// Persist the file to disk.
string outputPath = Path.Combine(reportsDir, $"Summary_{index}.docx");
document.Save(outputPath);

//ExStep:7-
}
//ExEnd
