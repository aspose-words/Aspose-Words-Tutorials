//ExSummary: Ez a bemutató azt mutatja be, hogyan lehet Word dokumentumot létrehozni, oszlopdiagramot beszúrni, mintafeladatokkal feltölteni, elrejteni a diagram ábrázolási területének keretét miközben a tengelyvonalak láthatóak maradnak, és végül menteni a dokumentumot.
//ExStepSummary:0: Hozzon létre egy új üres Word dokumentumot és egy DocumentBuilder objektumot a szerkesztéséhez.
//ExStepSummary:1: Szúrjon be egy 432×252 méretű oszlopdiagramot a dokumentumba, és szerezze meg a hozzá tartozó Chart objektumot.
//ExStepSummary:2: Törölje a meglévő sorozatokat, definiáljon három kategóriacímkét, és adjon hozzá egy sorozatot mintafeladatokkal a diagramhoz.
//ExStepSummary:3: Állítsa a plot terület vonalvastagságát nullára, így eltávolítva a keretet, miközben a tengelyvonalak láthatóak maradnak.
//ExStepSummary:4: Mentse a módosított dokumentumot .docx fájlba.
//ExStepImage:1:images/1.png
//ExStepImage:2:images/2.png
//ExStepImage:3:images/3.png
//ExStep:0-
Document wordDoc = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(wordDoc);

//ExStep:1-
Shape chartContainer = docBuilder.InsertChart(ChartType.Column, 432, 252);
Chart columnChart = chartContainer.Chart;

//ExStep:2-
columnChart.Series.Clear();
string[] labels = { "Group A", "Group B", "Group C" };
columnChart.Series.Add("Demo Data", labels, new double[] { 10, 20, 30 });

//ExStep:3-
columnChart.Format.Stroke.Weight = 0;

//ExStep:4-
wordDoc.Save("ChartWithoutPlotBorder.docx");
//ExEnd
