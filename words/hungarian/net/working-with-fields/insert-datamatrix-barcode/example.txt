//ExSummary: Ez a bemutató azt mutatja be, hogyan lehet új Word dokumentumot létrehozni, egy DISPLAYBARCODE mezőt beszúrni, azt DataMatrix vonalkódra konfigurálni, és a vonalkód képet a Document és a DocumentBuilder osztályok segítségével megjeleníteni.
//ExStepSummary:0: Hozzon létre egy új üres Word Document-et és egy DocumentBuilder-t a szerkesztéséhez.
//ExStepSummary:1: Szúrjon be egy DISPLAYBARCODE mezőt az aktuális kurzorpozícióba, amely mezőhelyőrzőt ad a dokumentumhoz.
//ExStepSummary:2: Állítsa be a mező BarcodeType értékét DataMatrix-re, és adja meg a kódolandó adatkarakterláncot.
//ExStepSummary:3: Opcionálisan definiálhatja a vonalkód háttér- és előtérszíneit.
//ExStepSummary:4: Hívja meg a dokumentumon az UpdateFields metódust a vonalkód kép mezőn belüli megjelenítéséhez.
//ExStepSummary:5: Mentse a dokumentumot .docx fájlba.
//ExStepImage:4:images/1.png
//ExStep:0-
// Initialise an empty Word document.
Document document = new Document();
DocumentBuilder docBuilder = new DocumentBuilder(document);

//ExStep:1-
// Add a DISPLAYBARCODE field to the document.
FieldDisplayBarcode displayBarcodeField = (FieldDisplayBarcode)docBuilder.InsertField(FieldType.FieldDisplayBarcode, true);

//ExStep:2-
// Define the barcode format as DataMatrix and assign a sample payload.
displayBarcodeField.BarcodeType = "DataMatrix";
displayBarcodeField.BarcodeValue = "9876543210";

//ExStep:3-
// (Optional) Specify background and foreground colours.
displayBarcodeField.BackgroundColor = "0xFFFFFF";
displayBarcodeField.ForegroundColor = "0x000000";

//ExStep:4-
// Refresh all fields so the barcode image is rendered.
document.UpdateFields();

//ExStep:5-
// Persist the document to a file.
document.Save("DataMatrix_BarcodeDemo.docx");
//ExEnd
