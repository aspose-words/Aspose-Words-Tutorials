//ExStepSummary:0: Adja meg a mappát, ahová a létrehozott dokumentumot menteni kell.
//ExStepSummary:1: Hozzon létre egy új üres Document objektumot és egy hozzá kapcsolódó DocumentBuilder-t a szerkesztéshez.
//ExStepSummary:2: Szúrjon be egy TC mezőt a "Entry Text" bejegyzésszöveggel, amely a tartalomjegyzékhez használható.
//ExStepSummary:3: Mentse a módosított dokumentumot a megadott könyvtárba.
//ExStepImage:2:images/1.png
//ExStep:0-
string dataDir = "YOUR DOCUMENT DIRECTORY";

//ExStep:1-
Document doc = new Document();
DocumentBuilder builder = new DocumentBuilder(doc);

//ExStep:2-
builder.InsertField("TC \"Entry Text\" \\f t");

//ExStep:3-
doc.Save(dataDir + "AddContentUsingDocumentBuilder.InsertTCField.docx");
